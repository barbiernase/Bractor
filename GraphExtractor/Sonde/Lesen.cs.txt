namespace Leihwesen.Lesen;

public class LeihEintrag : IReadModel
{
    public Guid Id { get; set; }
    public Guid LeserId { get; set; }
    public bool Offen { get; set; }
}

// Geteiltes Dokument (Marker IGeteiltesReadModel): mehrere Streams schreiben hinein — frei benannt, ohne Store-Kante.
public record Ausleihzaehler : IReadModel, IGeteiltesReadModel
{
    public string Id { get; init; } = "gesamt";
    public int Offen { get; init; }
}

// Fähigkeiten: frei benannt, je genau EINE Funktion; das Bündel (IStore) ist der Store.
public interface INotiereAusleihe : IWriteStore { Task SchreibeAsync(LeihEintrag eintrag); }
public interface ISchliesseAusleihe : IWriteStore { Task SchliesseAsync(Guid id); }
public interface IZaehleOffene : IReadStore { Task<IReadOnlyList<LeihEintrag>> OffeneAsync(Guid leserId); }

public interface ILeihBuch : IStore, INotiereAusleihe, ISchliesseAusleihe, IZaehleOffene { }

public sealed class LeihBuchSpeicher : ILeihBuch
{
    private readonly Dictionary<Guid, LeihEintrag> _daten = new();

    public Task SchreibeAsync(LeihEintrag eintrag) { _daten[eintrag.Id] = eintrag; return Task.CompletedTask; }

    public Task SchliesseAsync(Guid id) { if (_daten.TryGetValue(id, out var e)) e.Offen = false; return Task.CompletedTask; }

    public Task<IReadOnlyList<LeihEintrag>> OffeneAsync(Guid leserId) =>
        Task.FromResult<IReadOnlyList<LeihEintrag>>(_daten.Values.Where(e => e.Offen && e.LeserId == leserId).ToList());
}

public record OffeneAusleihen : IQuery
{
    public Guid LeserId { get; init; }
}

public record OffeneAusleihenAntwort(int Anzahl) : IQueryResponse;
public record KeineAusleihen() : IQueryResponse;

public partial class Leihliste : ISubscriber, IPullSubscriber
{
    private const string Kennung = "leihliste";

    public string SubscriberId => Kennung;

    // Verschleierung 1: die Fähigkeit geht durch eine Hilfsmethode.
    public async Task Handle(Ausgeliehen evt, IAggregateEnvelope envelope, ProjectionWriter writer, INotiereAusleihe notiere)
    {
        await writer.Execute(envelope.AggregateId.ToString(), async ctx =>
        {
            ctx.Track<Leihvorgang>(envelope.AggregateId);
            await Notiere(notiere, envelope.AggregateId, evt.LeserId);
        });
    }

    private static Task Notiere(INotiereAusleihe f, Guid id, Guid leser) =>
        f.SchreibeAsync(new LeihEintrag { Id = id, LeserId = leser, Offen = true });

    // Verschleierung 2+3: über einen Delegate und über ein fremdes Objekt; dazu ein nie ausgeführtes Lambda.
    public async Task Handle(Zurueckgegeben evt, IAggregateEnvelope envelope, ProjectionWriter writer, ISchliesseAusleihe schliesse)
    {
        Func<Guid, Task> tu = schliesse.SchliesseAsync;
        Func<Task> nie = () => new Schliesser(schliesse).Los(Guid.Empty);
        await writer.Execute(envelope.AggregateId.ToString(), async ctx =>
        {
            ctx.Track<Leihvorgang>(envelope.AggregateId);
            await tu(envelope.AggregateId);
        });
    }

    private sealed class Schliesser(ISchliesseAusleihe f) { public Task Los(Guid id) => f.SchliesseAsync(id); }
}

[ProjectionReader(TrackDeps = false)]
public partial class LeihAuskunftLeser : IReader<Leihliste>
{
    public async Task<OneOf<OffeneAusleihenAntwort, KeineAusleihen>> Handle(OffeneAusleihen query, IMessageEnvelope envelope, ReadContext ctx, IZaehleOffene zaehle)
    {
        var offen = await zaehle.OffeneAsync(query.LeserId);
        if (offen.Count == 0) return new KeineAusleihen();
        return new OffeneAusleihenAntwort(offen.Count);
    }
}

public partial class MahnungBeiSchaden : ISubscriber, IPullSubscriber
{
    public string SubscriberId => "mahnung-bei-schaden";

    public async IAsyncEnumerable<EroeffneMahnung> Handle(SchadenGemeldet evt, IAggregateEnvelope envelope, ProjectionWriter writer)
    {
        await Task.CompletedTask;
        yield return new EroeffneMahnung(Guid.NewGuid(), 5m);
    }
}

// Akteure: frei benannt; die Befugnis steht NUR in der Basisliste (IDarf<T>) — Reihenfolge = Deklaration. Kein Name verrät
// die Rolle (keine *Akteur-/*Rolle-Endung); eine Query zu dürfen ist dasselbe Wort wie ein Command.
public sealed record Theke : IAkteur, IDarf<LeiheAus>, IDarf<GibZurueck>, IDarf<OffeneAusleihen>;

// Eine KI als Akteur (frei benannt, Art in der Basisliste).
public sealed record Gutachterin : IKi, IDarf<EroeffneMahnung>;

// Ein Akteur, der DRAUSSEN reagiert (Vertrag, docs/konzept-akteure.md §3): frei benannt — weder der Akteur noch sein Vertrag
// verraten über den Namen, was zusammengehört (IAkteurVertrag<A> ist der einzige Anker). Ein Akteur ohne IDarf: was er als
// Zusage hineingibt, steht in der Signatur — einzeln (konkreter Typ), als Strom, und eine Zusage nur zur Kenntnis.
public sealed record Archiv : IMaschine;
public interface IRegal : IAkteurVertrag<Archiv>
{
    MeldeSchaden Auf(Zurueckgegeben e);
    IAsyncEnumerable<OneOf<EroeffneMahnung, MeldeSchaden>> Auf(SchadenGemeldet e);
    void Auf(Ausgeliehen e);
}

// Clients (docs/konzept-akteure.md §4): frei benannt — IClientVertrag ist der einzige Anker. EIN Client verkörpert ZWEI Akteure
// (das Archiv über seinen Vertrag, die Theke über ISendet/IFragt), und ein Akteur läuft über ZWEI Clients (das Archiv: Lesesaal + Depot).
// Eigene Kenntnis (void Auf) gehört dem Client, nicht einem Akteur.
public interface ILesesaal : IClientVertrag, IRegal, ISendet<LeiheAus>, IFragt<OffeneAusleihen>
{
    void Auf(MahnungEroeffnet e);
}
public interface IDepot : IClientVertrag, IRegal { }
