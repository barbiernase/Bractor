using Abstractions;

// SONDE — ABSICHTLICH FALSCH (docs/04 §4.4): eine Funktion und ihr Graph in der Anwendung. Eine Anwendung hat keine Funktionen und

namespace Leihwesen.Erinnerung;

public record Erinnere(Guid LeserId) : ICommand;
public record Erinnert(Guid LeserId) : ITransientEvent;

public sealed class Erinnerungsdienst : IFunktion
{
    public async Task<Erinnert> Auf(Erinnere e) => new(e.LeserId);
}

public sealed record Erinnerungslauf([Kopf] Erinnerungsdienst Dienst) : IGraph;
