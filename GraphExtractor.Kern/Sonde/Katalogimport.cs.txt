using Abstractions;
using Core;

// SONDE — fremdes Szenario 3 (Plan Pub/Sub-Neustrukturierung §6, Phase 0): MASSEN-IMPORT MIT FORTSCHRITT.
//
// EINE Quell-Nachricht (eine abgelegte Katalog-Datei) wird zu vielen Fakten: der Import-Vorgang ist ein Aggregat, ein Import-Dienst draußen
// liest die Datei und gibt als STROM je Zeile einen Artikel und zwischendurch den Fortschritt aus; die Leseseite zeigt den Stand. Heute ist
// der Strom eine Zusage eines Akteur-Vertrags (getragen von einem Client); im Zielbild eine Funktion mit Strom-Ausgabe (Plan §2.2, ITraegt).

namespace Warenkatalog.Import;

// Die Quelle: der Datei-Treiber befüllt die geerbten Felder, die Kennung sagt die Konfig.
public sealed record KatalogDateiAbgelegt : DateiGemeldet;

public record StarteImport(Guid AggregateId, string Pfad) : ICreationCommand;
public record ImportGestartet(Guid ImportId, string Pfad) : IEvent;
public record ImportLaeuftSchon(Guid ImportId) : ITransientEvent;

public record MeldeImportFortschritt(Guid AggregateId, int Zeilen) : ICommand;
public record ImportFortgeschritten(int Zeilen) : IEvent;

public record SchliesseImportAb(Guid AggregateId, int Zeilen, int Fehler) : ICommand;
public record ImportAbgeschlossen(int Zeilen, int Fehler) : IEvent;

public partial class Importlauf : IState
{
    public int Zeilen { get; set; }
    public bool Fertig { get; set; }

    public partial class Decider : IDecider<Importlauf>
    {
        public IEnumerable<OneOf<ImportGestartet, ImportLaeuftSchon>> Decide(StarteImport c)
        {
            if (this.State.Version > 0) { yield return new ImportLaeuftSchon(c.AggregateId); yield break; }
            yield return new ImportGestartet(c.AggregateId, c.Pfad);
        }

        // Fortschritt zählt nur vorwärts: eine verspätete oder wiederholte Meldung ändert nichts.
        public IEnumerable<OneOf<ImportFortgeschritten>> Decide(MeldeImportFortschritt c)
        {
            if (this.State.Fertig || c.Zeilen <= this.State.Zeilen) yield break;
            yield return new ImportFortgeschritten(c.Zeilen);
        }

        public IEnumerable<OneOf<ImportAbgeschlossen>> Decide(SchliesseImportAb c)
        {
            if (this.State.Fertig) yield break;
            yield return new ImportAbgeschlossen(c.Zeilen, c.Fehler);
        }
    }

    public partial class Applier : IApplier<Importlauf>
    {
        public void Apply(ImportGestartet e) { }
        public void Apply(ImportFortgeschritten e) { this.State.Zeilen = e.Zeilen; }
        public void Apply(ImportAbgeschlossen e) { this.State.Zeilen = e.Zeilen; this.State.Fertig = true; }
    }
}

public record LegeArtikelAn(Guid AggregateId, string Nummer, string Bezeichnung, decimal Preis) : ICreationCommand;
public record ArtikelAngelegt(string Nummer, string Bezeichnung, decimal Preis) : IEvent;
public record ArtikelExistiert(Guid ArtikelId) : ITransientEvent;

public partial class Artikel : IState
{
    public string Nummer { get; set; } = "";
    public decimal Preis { get; set; }

    public partial class Decider : IDecider<Artikel>
    {
        public IEnumerable<OneOf<ArtikelAngelegt, ArtikelExistiert>> Decide(LegeArtikelAn c)
        {
            if (this.State.Version > 0) { yield return new ArtikelExistiert(c.AggregateId); yield break; }
            yield return new ArtikelAngelegt(c.Nummer, c.Bezeichnung, c.Preis);
        }
    }

    public partial class Applier : IApplier<Artikel>
    {
        public void Apply(ArtikelAngelegt e) { this.State.Nummer = e.Nummer; this.State.Preis = e.Preis; }
    }
}

// Die Quell-Nachricht startet den Import-Vorgang; seine Id ist die des Vorgangs-Streams der Quelle (Strom).
public sealed class Katalogeingang : IPipeline
{
    public PipelineFluss Fluss => PipelineFluss.Definiere(p =>
    {
        var datei = p.Quelle<KatalogDateiAbgelegt>();
        var start = p.Alle(datei, datei.Strom()).Sende<StarteImport>((d, strom) => new StarteImport(strom.Id, d.Pfad));
    });
}

// Der Strom: je Zeile ein Artikel, zwischendurch der Fortschritt, am Ende der Abschluss — eine Zusage des Import-Dienstes draußen.
public sealed record Importdienst : IMaschine;
public interface IImportLauf : IAkteurVertrag<Importdienst>
{
    IAsyncEnumerable<OneOf<LegeArtikelAn, MeldeImportFortschritt, SchliesseImportAb>> Auf(ImportGestartet e);
}

// Leseseite: der Stand je Import.
public class Importstand : IReadModel
{
    public Guid Id { get; set; }
    public string Pfad { get; set; } = "";
    public int Zeilen { get; set; }
    public bool Fertig { get; set; }
}

public interface IBeginneImportstand : IWriteStore { Task BeginneAsync(Guid importId, string pfad); }
public interface ISetzeImportstand : IWriteStore { Task SetzeAsync(Guid importId, int zeilen, bool fertig); }
public interface IFindeImportstand : IReadStore { Task<Importstand?> FindeAsync(Guid importId); }
public interface IImportUebersicht : IStore, IBeginneImportstand, ISetzeImportstand, IFindeImportstand { }

public partial class Importfortschritt : ISubscriber, IPullSubscriber
{
    public string SubscriberId => "importfortschritt";

    public async Task Handle(ImportGestartet evt, IAggregateEnvelope envelope, ProjectionWriter writer, IBeginneImportstand beginne)
    {
        await writer.Execute(envelope.AggregateId.ToString(), ctx => beginne.BeginneAsync(envelope.AggregateId, evt.Pfad));
    }

    public async Task Handle(ImportFortgeschritten evt, IAggregateEnvelope envelope, ProjectionWriter writer, ISetzeImportstand setze)
    {
        await writer.Execute(envelope.AggregateId.ToString(), ctx => setze.SetzeAsync(envelope.AggregateId, evt.Zeilen, false));
    }

    public async Task Handle(ImportAbgeschlossen evt, IAggregateEnvelope envelope, ProjectionWriter writer, ISetzeImportstand setze)
    {
        await writer.Execute(envelope.AggregateId.ToString(), ctx => setze.SetzeAsync(envelope.AggregateId, evt.Zeilen, true));
    }
}

public record HoleImportstand(Guid ImportId) : IQuery;
public record ImportstandAntwort(int Zeilen, bool Fertig) : IQueryResponse;
public record ImportUnbekannt() : IQueryResponse;

[ProjectionReader(TrackDeps = false)]
public partial class Importauskunft : IReader<Importfortschritt>
{
    public async Task<OneOf<ImportstandAntwort, ImportUnbekannt>> Handle(HoleImportstand query, IMessageEnvelope envelope, ReadContext ctx, IFindeImportstand finde)
    {
        var stand = await finde.FindeAsync(query.ImportId);
        if (stand is null) return new ImportUnbekannt();
        return new ImportstandAntwort(stand.Zeilen, stand.Fertig);
    }
}

// Wer hineingibt: die Ablage (Maschine) meldet die Datei; die Katalogpflege fragt den Stand und sieht den Fortschritt.
public sealed record Ablage : IMaschine, IDarf<KatalogDateiAbgelegt>;
public sealed record Katalogpflege : IMensch, IDarf<HoleImportstand>;

public interface IImportWorker : IClientVertrag, IImportLauf { }

public interface IKatalogpflegeOberflaeche : IClientVertrag, IFragt<HoleImportstand>
{
    void Auf(ImportFortgeschritten e);
    void Auf(ImportAbgeschlossen e);
}
