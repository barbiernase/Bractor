using System.IO;
using System.Text;
using Abstractions;
using Core;
using Core.Katalog;

// SONDE — fremdes Szenario 3 (docs/12 §12.6): MASSEN-IMPORT MIT FORTSCHRITT.
//
// EINE abgelegte Katalog-Datei wird zu vielen Fakten. Am Kopf des Graphen steht eine QUELLFUNKTION AUS DEM KATALOG des Frameworks (der
// Dateiwächter, Core.Katalog): je gefundene Datei ein Lauf. Der Eingang startet den Import-Vorgang (ein Aggregat) und hört dessen Event als Antwort;
// ein Zeilenleser (Technik als KLASSE, ohne Schnittstelle, mit Konfig-Record) gibt als STROM je Zeile einen Unterlauf (JE) und zwischendurch den
// Fortschritt an den Vorgang; je Zeile wird ein Artikel angelegt; die Bilanz SAMMELT alle Zeilen und JOINT sie mit dem bereiten Katalog derselben
// Ebene. Die Ids leiten die Knoten aus der Herkunft ab (R13). Der Akteur der Instanz (die Ablage) darf, was den Graphen verlässt.
//
// Die Leseseite zeigt die beiden Store-Klassen ohne Transaktion (docs/04 §4.3, docs/10 R14): der Stand liegt in Dateien (Klasse 2, wiederholungsfest — jede
// Schreib-Fähigkeit nimmt die Herkunft und schreibt bedingt), und eine Metrik zählt angelegte Artikel (Klasse 3, mindestens einmal — nur mit
// ausdrücklicher Erlaubnis an der Projektion).

namespace Warenkatalog.Import;

public record StarteImport(Id<Importlauf> AggregateId, string Pfad) : ICommand<Importlauf>, ICreationCommand;
public record ImportGestartet(Id<Importlauf> ImportId, string Pfad) : IEvent;
public record ImportLaeuftSchon(Id<Importlauf> ImportId) : ITransientEvent;

public record MeldeImportFortschritt(Id<Importlauf> AggregateId, int Zeilen) : ICommand<Importlauf>;
public record ImportFortgeschritten(int Zeilen) : IEvent;

public record SchliesseImportAb(Id<Importlauf> AggregateId, int Zeilen, int Fehler) : ICommand<Importlauf>;
public record ImportAbgeschlossen(int Zeilen, int Fehler) : IEvent;

public partial class Importlauf : IState
{
    public int Zeilen { get; set; }
    public bool Fertig { get; set; }

    public partial class Decider : IDecider<Importlauf>
    {
        public IEnumerable<OneOf<ImportGestartet, ImportLaeuftSchon>> Decide(StarteImport c)
        {
            if (this.State.Version > 0) { yield return new ImportLaeuftSchon(c.AggregateId); yield break; }
            yield return new ImportGestartet(c.AggregateId, c.Pfad);
        }

        // Fortschritt zählt nur vorwärts: eine verspätete oder wiederholte Meldung ändert nichts.
        public IEnumerable<OneOf<ImportFortgeschritten>> Decide(MeldeImportFortschritt c)
        {
            if (this.State.Fertig || c.Zeilen <= this.State.Zeilen) yield break;
            yield return new ImportFortgeschritten(c.Zeilen);
        }

        public IEnumerable<OneOf<ImportAbgeschlossen>> Decide(SchliesseImportAb c)
        {
            if (this.State.Fertig) yield break;
            yield return new ImportAbgeschlossen(c.Zeilen, c.Fehler);
        }
    }

    public partial class Applier : IApplier<Importlauf>
    {
        public void Apply(ImportGestartet e) { }
        public void Apply(ImportFortgeschritten e) { this.State.Zeilen = e.Zeilen; }
        public void Apply(ImportAbgeschlossen e) { this.State.Zeilen = e.Zeilen; this.State.Fertig = true; }
    }
}

public record LegeArtikelAn(Id<Artikel> AggregateId, string Nummer, string Bezeichnung, decimal Preis) : ICommand<Artikel>, ICreationCommand;
public record ArtikelAngelegt(string Nummer, string Bezeichnung, decimal Preis) : IEvent;
public record ArtikelExistiert(Id<Artikel> ArtikelId) : ITransientEvent;

public partial class Artikel : IState
{
    public string Nummer { get; set; } = "";
    public decimal Preis { get; set; }

    public partial class Decider : IDecider<Artikel>
    {
        public IEnumerable<OneOf<ArtikelAngelegt, ArtikelExistiert>> Decide(LegeArtikelAn c)
        {
            if (this.State.Version > 0) { yield return new ArtikelExistiert(c.AggregateId); yield break; }
            yield return new ArtikelAngelegt(c.Nummer, c.Bezeichnung, c.Preis);
        }
    }

    public partial class Applier : IApplier<Artikel>
    {
        public void Apply(ArtikelAngelegt e) { this.State.Nummer = e.Nummer; this.State.Preis = e.Preis; }
    }
}

// ── Der Graph: Dateiwächter (Katalog) → Eingang → Zeilen (Je) → Artikel; Bilanz (Sammeln + Join) ───────────────────────────────────────────────
public record KatalogBereit(Id<Importlauf> Import, string Pfad) : ICommand;
public record Zeile(Id<Importlauf> Import, int Nummer, string[] Felder) : ICommand;

// Der Eingang: aus der gefundenen Datei der Import-Vorgang (Id aus der Herkunft); seine Antwort macht den Katalog bereit.
public sealed class Katalogeingang : IFunktion
{
    public StarteImport Auf(DateiGefunden d, Herkunft h) => new(h.NeueId<Importlauf>(d.RelativerPfad), d.Pfad);

    public KatalogBereit Auf(ImportGestartet e) => new(e.ImportId, e.Pfad);
}

// Technik als Klasse (ohne Schnittstelle, ohne Marker) mit Konfig-Record.
public record CsvKonfig(string Trenner = ";");

public sealed class CsvZerleger(CsvKonfig konfig)
{
    public string[] Zerlege(string zeile) => zeile.Split(konfig.Trenner);
}

// JE: je Zeile ein Unterlauf; je hundert Zeilen der Fortschritt an den Vorgang (er verlässt den Graphen).
public sealed class Zeilenleser(CsvZerleger zerleger) : IFunktion
{
    public async IAsyncEnumerable<OneOf<Zeile, MeldeImportFortschritt>> Auf(KatalogBereit k)
    {
        var nummer = 0;
        await foreach (var text in File.ReadLinesAsync(k.Pfad))
        {
            nummer++;
            yield return new Zeile(k.Import, nummer, zerleger.Zerlege(text));
            if (nummer % 100 == 0) yield return new MeldeImportFortschritt(k.Import, nummer);
        }
    }
}

// Im Unterlauf je Zeile: der Artikel (Id aus der Herkunft und der Artikelnummer).
public sealed class Artikelanlage : IFunktion
{
    public LegeArtikelAn Auf(Zeile z, Herkunft h) =>
        new(h.NeueId<Artikel>(z.Felder[0]), z.Felder[0], z.Felder.ElementAtOrDefault(1) ?? "", decimal.Parse(z.Felder.ElementAtOrDefault(2) ?? "0"));
}

// SAMMELN + JOIN: alle Zeilen der Unterläufe und der bereite Katalog auf der Ebene des Kopfs.
public sealed class Importbilanz : IFunktion
{
    public SchliesseImportAb Auf(IReadOnlyList<Zeile> alle, KatalogBereit katalog) =>
        new(katalog.Import, alle.Count, alle.Count(z => z.Felder.Length < 3));
}

public sealed record Katalogimport(Dateiwaechter Eingang, Katalogeingang Annahme, Zeilenleser Lesen, Artikelanlage Anlage, Importbilanz Bilanz) : IGraph;

// Leseseite: der Stand je Import.
public class Importstand : IReadModel
{
    public Guid Id { get; set; }
    public string Pfad { get; set; } = "";
    public int Zeilen { get; set; }
    public bool Fertig { get; set; }
}

public interface IBeginneImportstand : IWriteStore { Task BeginneAsync(Guid importId, string pfad, Herkunft h); }
public interface ISetzeImportstand : IWriteStore { Task SetzeAsync(Guid importId, int zeilen, bool fertig, Herkunft h); }
public interface IFindeImportstand : IReadStore { Task<Importstand?> FindeAsync(Guid importId); }
public interface IImportUebersicht : IStore, IBeginneImportstand, ISetzeImportstand, IFindeImportstand { }

// Klasse 2 über die Basis (DateiZiel : IWiederholungsfestesZiel): je Import zwei Dateien, jede nur geschrieben, wenn die Version des Events
// größer ist als die gespeicherte — eine Wiederholung oder eine zweite Aktivierung mit älterem Stand ergibt dasselbe.
public sealed class Importablage : DateiZiel, IImportUebersicht
{
    public Importablage(StoreOrte orte) : base(orte.Verzeichnis(nameof(Importablage))) { }

    public Task BeginneAsync(Guid importId, string pfad, Herkunft h) => LegeAbWennNeuer($"import/{importId}/pfad", Encoding.UTF8.GetBytes(pfad), h);

    public Task SetzeAsync(Guid importId, int zeilen, bool fertig, Herkunft h) =>
        LegeAbWennNeuer($"import/{importId}/stand", Encoding.UTF8.GetBytes($"{zeilen};{fertig}"), h);

    public Task<Importstand?> FindeAsync(Guid importId)
    {
        if (Lies($"import/{importId}/pfad") is not { } pfad) return Task.FromResult<Importstand?>(null);
        var stand = Lies($"import/{importId}/stand") is { } s ? Encoding.UTF8.GetString(s).Split(';') : ["0", "False"];
        return Task.FromResult<Importstand?>(new Importstand
        {
            Id = importId, Pfad = Encoding.UTF8.GetString(pfad), Zeilen = int.Parse(stand[0]), Fertig = bool.Parse(stand[1]),
        });
    }
}

public sealed class Importfortschritt : IProjektion
{
    public Task Auf(ImportGestartet e, Herkunft h, IBeginneImportstand beginne) => beginne.BeginneAsync(h.Stream, e.Pfad, h);

    public Task Auf(ImportFortgeschritten e, Herkunft h, ISetzeImportstand setze) => setze.SetzeAsync(h.Stream, e.Zeilen, false, h);

    public Task Auf(ImportAbgeschlossen e, Herkunft h, ISetzeImportstand setze) => setze.SetzeAsync(h.Stream, e.Zeilen, true, h);
}

// Klasse 3 über die Basis (DateiSink : INurMindestensEinmal): je angelegtem Artikel eine Zeile — nach einem Abbruch kann sie doppelt stehen.
public interface IProtokolliereArtikel : IWriteStore { Task ProtokolliereAsync(string nummer, decimal preis); }
public interface IArtikelprotokoll : IStore, IProtokolliereArtikel { }

public sealed class Artikelprotokoll : DateiSink, IArtikelprotokoll
{
    public Artikelprotokoll(StoreOrte orte) : base(orte.Verzeichnis(nameof(Artikelprotokoll))) { }

    public Task ProtokolliereAsync(string nummer, decimal preis) => HaengeAn("artikel.log", $"{nummer};{preis}");
}

// Die ausdrückliche Erlaubnis, in einen Store der Klasse 3 zu schreiben; ohne Herkunft, mit Abbruch-Parameter.
[MindestensEinmal]
public sealed class Artikelzaehlung : IProjektion
{
    public Task Auf(ArtikelAngelegt e, IProtokolliereArtikel protokoll, CancellationToken ct) => protokoll.ProtokolliereAsync(e.Nummer, e.Preis);
}

public record HoleImportstand(Id<Importlauf> ImportId) : IQuery;
public record ImportstandAntwort(int Zeilen, bool Fertig) : IQueryResponse;
public record ImportUnbekannt() : IQueryResponse;

public sealed class Importauskunft : IReader
{
    public async Task<OneOf<ImportstandAntwort, ImportUnbekannt>> Frage(HoleImportstand q, IFindeImportstand finde)
    {
        var stand = await finde.FindeAsync(q.ImportId.Wert);
        if (stand is null) return new ImportUnbekannt();
        return new ImportstandAntwort(stand.Zeilen, stand.Fertig);
    }
}

// und hört den Fortschritt (Kenntnis).
public sealed record Ablage : IAkteur, IDarf<StarteImport>, IDarf<LegeArtikelAn>, IDarf<MeldeImportFortschritt>, IDarf<SchliesseImportAb>;
public sealed record Katalogpflege : IAkteur, IDarf<HoleImportstand>, IHoert<ImportFortgeschritten>, IHoert<ImportAbgeschlossen>;

public sealed record KatalogpflegeOberflaeche : IClient, IVerkoerpert<Katalogpflege>;
