using System.Threading.Tasks;
using Abstractions;

namespace Sondenkatalog.Katalog;

/// <summary>
/// Peilt eine Frequenz mit Technik und Konfig — eine WAHL: gepeilt oder ein Hinweis. Kommt die Bestätigung der Anwendung zurück (die Antwort ihres
/// Aggregats, übersetzt), meldet sie die Abweichung zur letzten Richtung — gelesen über eine Fähigkeit, die der Katalog deklariert.
/// </summary>
public sealed class Peilung(PeilKonfig konfig, IPeilempfaenger empfaenger) : IFunktion
{
    public async Task<Peilabweichung> Auf(PeilungBestaetigt b, IHoleLetzteRichtung letzte) => new(b.RichtungGrad - await letzte.LetzteAsync());

    public async Task<OneOf<Gepeilt, PeilungUnmoeglich>> Auf(Peile p)
    {
        var messung = await empfaenger.PeileAsync(p.FrequenzMhz, p.DauerMs);
        if (messung[1] < konfig.MindestGuete) return new PeilungUnmoeglich(p.FrequenzMhz, "Güte zu gering");
        return new Gepeilt(p.Auftrag, p.FrequenzMhz, messung[0], messung[1]);
    }
}
