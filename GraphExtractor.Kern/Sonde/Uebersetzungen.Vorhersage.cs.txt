using Abstractions;
using Sondenkatalog.Katalog;

// SONDE — die Übersetzungen des Wetterdiensts (docs/04 §4.4): der Kopf der Prognose aus der Nachricht des Warndiensts (die Station als
// Guid ausdrücklich, die Temperatur nach Namen), die Prognose des Katalogs als Command an die Station.

namespace Wetterdienst.Vorhersage;

#pragma warning disable CS9113

public sealed class BildePrognose_Bilde_1(BildePrognose source) : ITranslate<BildePrognose, Bilde>
{
    public Bilde Translate() => new(Station: source.Station.Wert, Grad: source.Grad);
}

public sealed class Prognostiziert_VermerkePrognose_1(Prognostiziert source) : ITranslate<Prognostiziert, VermerkePrognose>
{
    public VermerkePrognose Translate() => new(AggregateId: new(source.Station), Grad: source.Grad, Gelernt: source.Gelernt);
}
