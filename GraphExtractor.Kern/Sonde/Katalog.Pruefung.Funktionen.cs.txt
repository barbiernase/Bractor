using Abstractions;
using Sondenkatalog.Katalog.Gebuehren;

namespace Sondenkatalog.Katalog.Pruefung;

// Der Start: je gescannter Rückgabe ein Vorgang (Id aus der Herkunft); seine Antwort öffnet JE Seite einen Unterlauf.
public sealed class Pruefungsstart : IFunktion
{
    public async Task<Starte> Auf(Scan e, Herkunft h) => new(h.NeueId("pruefung"), e.LeserId, e.Seiten);

    public async IAsyncEnumerable<SeiteZuPruefen> Auf(Gestartet g)
    {
        for (var i = 0; i < g.Seiten.Count; i++) yield return new SeiteZuPruefen(g.Pruefung, i + 1, g.Seiten[i]);
    }
}

// Technik ohne Marker mit zwei Umsetzungen — welche, sagt die Instanz (Technik am Knoten).
public interface ISeitenleser
{
    bool IstBeschaedigt(string seite);
}

public sealed class Lichtpruefung : ISeitenleser
{
    public bool IstBeschaedigt(string seite) => seite.Contains('#');
}

public sealed class Handpruefung : ISeitenleser
{
    public bool IstBeschaedigt(string seite) => seite.Length == 0;
}

// Im Unterlauf je Seite.
public sealed class Seitenpruefer(ISeitenleser leser) : IFunktion
{
    public async Task<Seitenbefund> Auf(SeiteZuPruefen s) => new(s.Pruefung, s.Nummer, leser.IstBeschaedigt(s.Seite));
}

// Lese-Fähigkeit: die Mahnung nach Ablauf der Leihfrist richtet sich nach der Zahl offener Ausleihen des Lesers.
public sealed class Offenzaehlung : IFunktion
{
    public async IAsyncEnumerable<Mahnbetrag> Auf(Bemesse b, Herkunft h, IZaehleOffeneAusleihen offene)
    {
        var anzahl = await offene.AnzahlAsync(b.LeserId);
        if (anzahl > 0) yield return new Mahnbetrag(h.NeueId("mahnung"), anzahl * 2m);
    }
}
