using Abstractions;
using Leihwesen.Ausleihe.Befehle;

// SONDE — Aggregat mit Commands in eigenem Sub-Namespace, Property-Records, if/else-Guards, alles in EINER Datei. Die Ausleihe setzt eine
//         FRIST (docs/10 §10.2): die Uhr schickt nach Ablauf der Leihdauer die Mahnung an DENSELBEN Leihvorgang; ist er schon zurück, wirkt sie nicht.
//         Der Leihvorgang beantwortet eine Query DIREKT (Frage im Decider, docs/03 §3.2): Ziel ist das Glied Id<Leihvorgang>, die Antwort ein OneOf.

namespace Leihwesen.Ausleihe.Befehle
{
    /// <summary>Leiht ein Medium aus.</summary>
    public record LeiheAus(Id<Leihvorgang> AggregateId, Signatur Medium, Guid LeserId, int Tage = 14) : ICommand<Leihvorgang>, ICreationCommand;

    public record GibZurueck : ICommand<Leihvorgang>
    {
        public Id<Leihvorgang> AggregateId { get; init; }
        public Zustand Zustand { get; init; }
    }

    public record MahneRueckgabe(Id<Leihvorgang> AggregateId) : ICommand<Leihvorgang>;
}

namespace Leihwesen.Ausleihe
{
    public enum Zustand { Neu = 1, Gebraucht = 2, Beschaedigt = 3 }

    public record Signatur(string Regal, int Nummer);

    public record Ausgeliehen(Signatur Medium, Guid LeserId, int Tage) : IEvent;

    public record Zurueckgegeben : IEvent
    {
        public Guid LeserId { get; init; }
        public Zustand Zustand { get; init; }
    }

    public record RueckgabeAngemahnt(Guid LeserId) : IEvent;

    public record BereitsAusgeliehen(Id<Leihvorgang> AusleiheId) : ITransientEvent;
    public record NichtAusgeliehen(Id<Leihvorgang> AusleiheId) : ITransientEvent;

    public record HoleLeihstand(Id<Leihvorgang> Vorgang) : IQuery;
    public record Leihstand(Id<Leihvorgang> Vorgang, Guid LeserId) : IQueryResponse;
    public record NichtVerliehen(Id<Leihvorgang> Vorgang) : IQueryResponse;

    public partial class Leihvorgang : IState
    {
        public Guid LeserId { get; set; }
        public bool Offen { get; set; }

        public partial class Decider : IDecider<Leihvorgang>
        {
            public IEnumerable<OneOf<Ausgeliehen, BereitsAusgeliehen, Frist<MahneRueckgabe>>> Decide(LeiheAus befehl)
            {
                if (this.State.Offen)
                {
                    yield return new BereitsAusgeliehen(befehl.AggregateId);
                }
                else
                {
                    yield return new Ausgeliehen(befehl.Medium, befehl.LeserId, befehl.Tage);
                    yield return new Frist<MahneRueckgabe>(new MahneRueckgabe(befehl.AggregateId), TimeSpan.FromDays(befehl.Tage));
                }
            }

            public IEnumerable<OneOf<RueckgabeAngemahnt>> Decide(MahneRueckgabe befehl)
            {
                if (!this.State.Offen) yield break;
                yield return new RueckgabeAngemahnt(this.State.LeserId);
            }

            public IEnumerable<OneOf<Zurueckgegeben, NichtAusgeliehen>> Decide(GibZurueck befehl)
            {
                if (!this.State.Offen)
                {
                    yield return new NichtAusgeliehen(befehl.AggregateId);
                    yield break;
                }
                yield return new Zurueckgegeben { LeserId = this.State.LeserId, Zustand = befehl.Zustand };
            }

            public OneOf<Leihstand, NichtVerliehen> Frage(HoleLeihstand frage) =>
                this.State.Offen ? new Leihstand(frage.Vorgang, this.State.LeserId) : new NichtVerliehen(frage.Vorgang);
        }

        public partial class Applier : IApplier<Leihvorgang>
        {
            public void Apply(Ausgeliehen e)
            {
                this.State.LeserId = e.LeserId;
                this.State.Offen = true;
            }

            public void Apply(Zurueckgegeben e)
            {
                this.State.Offen = false;
            }

            public void Apply(RueckgabeAngemahnt e) { }
        }
    }

    // Hilfscode ohne Baustein-Rolle (Werkbank P7.4): wird eine Code-Karte — partial in zwei Teilen, beide gehören zur selben Karte.
    public static partial class Fristrechnung
    {
        public static DateTime Faellig(DateTime ab, int tage) => ab.AddDays(tage);
    }

    public static partial class Fristrechnung
    {
        public static bool Ueberfaellig(DateTime faellig, DateTime jetzt) => jetzt > faellig;
    }
}
