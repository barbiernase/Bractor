// SONDE — Leseseite mit FREIEN Namen (keine Namenskonvention), class-ReadModel, OneOf-Antwort, ein Store der Klasse 1 (atomar, Marten).
// Verträge von Auf und Frage: WAS aus der Signatur (auch eine deklarierte, nie ausgegebene Variante), WELCHE Store-Fns aus den
// Fähigkeits-Parametern, WELCHE Projektion hinter einem Reader aus seinen Lese-Fähigkeiten (derselbe Store). Verschleierung: Fähigkeit über
// eine Hilfsmethode, über einen Delegate, über ein fremdes Objekt, in einem nie ausgeführten Lambda — die Kanten bleiben exakt, weil sie nur
// aus der Signatur kommen. Kein Rumpf-Fakt (kein Guard).

namespace Leihwesen.Lesen;

public class LeihEintrag : IReadModel
{
    public Guid Id { get; set; }
    public Guid LeserId { get; set; }
    public bool Offen { get; set; }
}

// Geteiltes Dokument (Marker IGeteiltesReadModel): mehrere Streams schreiben hinein — frei benannt, ohne Store-Kante.
public record Ausleihzaehler : IReadModel, IGeteiltesReadModel
{
    public string Id { get; init; } = "gesamt";
    public int Offen { get; init; }
}

// Fähigkeiten: frei benannt, je genau EINE Funktion; das Bündel (IStore) ist der Store.
public interface INotiereAusleihe : IWriteStore { Task SchreibeAsync(LeihEintrag eintrag); }
public interface ISchliesseAusleihe : IWriteStore { Task SchliesseAsync(Guid id); }
public interface IZaehleOffene : IReadStore { Task<IReadOnlyList<LeihEintrag>> OffeneAsync(Guid leserId); }

public interface ILeihBuch : IStore, INotiereAusleihe, ISchliesseAusleihe, IZaehleOffene { }

// Klasse 1 über die Basis (MartenStore : IAtomaresZiel): Wirkungen und Fortschritt in einer Transaktion.
public sealed class LeihBuchSpeicher : MartenStore, ILeihBuch
{
    public LeihBuchSpeicher(IDocumentStore store) : base(store, typeof(LeihEintrag)) { }

    public Task SchreibeAsync(LeihEintrag eintrag) => EnqueueStore(eintrag.Id, eintrag);

    public Task SchliesseAsync(Guid id) => EnqueueTransform<LeihEintrag>(id, e => { e.Offen = false; return e; });

    public async Task<IReadOnlyList<LeihEintrag>> OffeneAsync(Guid leserId)
    {
        await using var session = Store.QuerySession();
        return await session.Query<LeihEintrag>().Where(e => e.Offen && e.LeserId == leserId).ToListAsync();
    }
}

public record OffeneAusleihen : IQuery
{
    public Guid LeserId { get; init; }
}

public record OffeneAusleihenAntwort(int Anzahl) : IQueryResponse;
public record KeineAusleihen() : IQueryResponse;

// Die Kennung ist der Typname — kein Feld, keine Konstante.
public sealed class Leihliste : IProjektion
{
    // Verschleierung 1: die Fähigkeit geht durch eine Hilfsmethode.
    public Task Auf(Ausgeliehen e, Herkunft h, INotiereAusleihe notiere) => Notiere(notiere, h.Stream, e.LeserId);

    private static Task Notiere(INotiereAusleihe f, Guid id, Guid leser) =>
        f.SchreibeAsync(new LeihEintrag { Id = id, LeserId = leser, Offen = true });

    // Verschleierung 2+3: über einen Delegate und über ein fremdes Objekt; dazu ein nie ausgeführtes Lambda.
    public async Task Auf(Zurueckgegeben e, Herkunft h, ISchliesseAusleihe schliesse)
    {
        Func<Guid, Task> tu = schliesse.SchliesseAsync;
        Func<Task> nie = () => new Schliesser(schliesse).Los(Guid.Empty);
        await tu(h.Stream);
    }

    private sealed class Schliesser(ISchliesseAusleihe f) { public Task Los(Guid id) => f.SchliesseAsync(id); }
}

// Der Reader nennt keine Projektion: dahinter steht, wer in denselben Store schreibt (Leihliste über ILeihBuch). Mit Abbruch-Parameter.
public sealed class LeihAuskunftLeser : IReader
{
    public async Task<OneOf<OffeneAusleihenAntwort, KeineAusleihen>> Frage(OffeneAusleihen q, IZaehleOffene zaehle, CancellationToken ct)
    {
        var offen = await zaehle.OffeneAsync(q.LeserId);
        if (offen.Count == 0) return new KeineAusleihen();
        return new OffeneAusleihenAntwort(offen.Count);
    }
}

// Akteure: frei benannt; was einer HINEINGIBT, steht als IDarf<T> in der Basisliste, was zu ihm KOMMT als IHoert<T> — Reihenfolge = Deklaration.
// Kein Name verrät die Rolle (keine *Akteur-/*Rolle-Endung); eine Query zu dürfen ist dasselbe Wort wie ein Command; ein Command ohne Ziel
// (die gescannte Rückgabe) geht an den Kopf eines Graphen. KENNTNIS: ein Hinweis aus einem Decide (NichtAusgeliehen), einer aus einem Knoten
// (NichtSchaetzbar) und einer des Frameworks (LaufGescheitert).
public sealed record Theke : IAkteur, IDarf<LeiheAus>, IDarf<GibZurueck>, IDarf<OffeneAusleihen>, IDarf<RueckgabeGescannt>,
    IHoert<NichtAusgeliehen>, IHoert<NichtSchaetzbar>, IHoert<LaufGescheitert>;

// Eine KI als Akteur (frei benannt, keine Art mehr): darf eine Mahnung eröffnen und hört, wenn eine eröffnet ist (ein Event: Kenntnis).
public sealed record Gutachterin : IAkteur, IDarf<EroeffneMahnung>, IHoert<MahnungEroeffnet>;

// ARBEIT (docs/konzept-akteure.md §5): das Archiv hört einen Command ohne Ziel aus einem Graphen (die Schadensaufnahme wählt die Begutachtung vor
// Ort); seine Antwort ist ein gewöhnlicher Command über sein IDarf. Dazu Kenntnis eines Events.
public sealed record Archiv : IAkteur, IHoert<BegutachteVorOrt>, IDarf<EroeffneMahnung>, IHoert<Ausgeliehen>;

// Clients (docs/konzept-akteure.md §4): frei benannt — ein Record mit IClient, der nur nennt, wen er verkörpert. EIN Client verkörpert ZWEI Akteure
// (die Theke und das Archiv, das Arbeit annimmt), und ein Akteur läuft über ZWEI Clients (das Archiv: Lesesaal + Depot). Was über die
// Leitung geht, sind das IDarf und IHoert der Akteure.
public sealed record Lesesaal : IClient, IVerkoerpert<Theke>, IVerkoerpert<Archiv>;
public sealed record Depot : IClient, IVerkoerpert<Archiv>;
