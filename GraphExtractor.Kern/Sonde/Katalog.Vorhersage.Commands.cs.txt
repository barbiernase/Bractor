using Abstractions;

// SONDE — fremder Katalog, Bereich Vorhersage (docs/plan-ohne-pipeline.md): die Prognose und die Kartenausgabe domänenneutral — eine Station ist ein
// Guid, keine Id einer Anwendung.

namespace Sondenkatalog.Katalog.Vorhersage;

/// <summary>Der Eingang der Prognose: eine Temperatur einer Station.</summary>
public record Bilde(Guid Station, double Grad) : ICommand;

// Was nur Knoten einander geben — auch generisch (das Typargument unterscheidet die beiden Glättungen).
public record Tendenz(Guid Station, double Grad) : ICommand;
public record Geglaettet<TFenster>(Guid Station, double Grad) : ICommand where TFenster : IFenster;

/// <summary>Das Ergebnis der Prognose: was das Modell für die Station gelernt hat — die Anwendung übersetzt es.</summary>
public record Prognostiziert(Guid Station, double Grad, int Gelernt) : ICommand;

/// <summary>Ein Command ohne Ziel an den Kopf der Kartenausgabe: die Karte einer Station ist zu zeichnen.</summary>
public record ZeichneKarte(Guid Station, double Grad) : ICommand;
/// <summary>Ein Command ohne Ziel an die Ablage (ein Kopf, der auch von innen nimmt): die Karte ist zu löschen.</summary>
public record LoescheKarte(Guid Station) : ICommand;
public record KarteGezeichnet(Guid Station, int Breite) : ICommand;
public record KarteSkaliert(Guid Station, int Breite) : ICommand;
