// Gleichbleibend: nur ein beschädigt zurückgegebenes Medium ist ein Schadensfall — je Rückgabe genau einer (Id aus der Herkunft).
public sealed class SchadenBeiRueckgabe : IFunktion
{
    public IEnumerable<Schaden> Auf(Rueckgabe p, Herkunft h)
    {
        if (p.Beschaedigt) yield return new Schaden(h.NeueId("schaden"), p.LeserId);
    }
}

// WAHL: genau einer der drei Fälle je Lauf.
public sealed class Schadensaufnahme : IFunktion
{
    public OneOf<SchadenErfasst, VorOrt, NichtSchaetzbar> Auf(Schaetze s)
    {
        if (s.LeserId == Guid.Empty) return new NichtSchaetzbar("unbekannter Leser");
        if (s.Fall.ToByteArray()[0] % 2 == 0) return new VorOrt(s.Fall, s.LeserId);   // an den Schadensfall
        return new SchadenErfasst(s.LeserId, 25m, 2);
    }
}

// Technik hinter der Schätzung — eine Schnittstelle ohne Marker (der Schätzer nimmt sie im Konstruktor), eine Umsetzung mit Konfig-Record
// (Werte am Knoten der Instanz).
public interface IWertTabelle
{
    /// <summary>Der Restwert nach linearer Abschreibung.</summary>
    decimal Restwert(decimal neuwert, int alterJahre);
    Task<bool> IstSchaetzbarAsync(Guid leserId);
}

public sealed class LineareWertTabelle(WertTabellenKonfig konfig) : IWertTabelle
{
    public decimal Restwert(decimal neuwert, int alterJahre) => Math.Max(konfig.Mindestwert, neuwert * (1 - alterJahre / (decimal)konfig.Abschreibungsjahre));

    public Task<bool> IstSchaetzbarAsync(Guid leserId)
    {
        return Task.FromResult(leserId != Guid.Empty);
    }
}

// KETTE, Schritt 1: die Schätzung mit Technik.
public sealed class Schadensschaetzer(IWertTabelle tabelle) : IFunktion
{
    public async Task<SchadenGeschaetzt> Auf(SchadenErfasst e)
    {
        var faktor = await tabelle.IstSchaetzbarAsync(e.LeserId) ? 1m : 0m;
        return new SchadenGeschaetzt(e.LeserId, faktor * tabelle.Restwert(e.Neuwert, e.AlterJahre));
    }
}

// KETTE, Schritt 2: ein Gutachten, das dauern kann — sein Zuschlag aus dem Konfig-Record des Knotens; den Betrag der Schätzung trägt es weiter.
public sealed class Gutachtenanfrage(GutachtenKonfig konfig) : IFunktion
{
    public async Task<GutachtenErhalten> Auf(SchadenGeschaetzt g)
    {
        await Task.Delay(konfig.Bedenkzeit);
        return new GutachtenErhalten(g.LeserId, g.Betrag, konfig.Zuschlag);
    }
}

// KETTE, Schritt 3: die Mahnung aus dem, was das Gutachten mitträgt.
public sealed class MahnungNachSchaetzung : IFunktion
{
    public Mahnbetrag Auf(GutachtenErhalten g, Herkunft h) => new(h.NeueId("mahnung"), g.Betrag + g.Zuschlag);
}

// Technik als Klasse (ohne Schnittstelle, ohne Marker) mit Konfig-Record.
public sealed class CsvZerleger(CsvKonfig konfig)
{
    public string[] Zerlege(string zeile) => zeile.Split(konfig.Trenner);
}

// JE: je Zeile ein Unterlauf; je hundert Zeilen der Fortschritt, am Ende die Bilanz (beide verlassen den Graphen).
public sealed class Zeilenleser(CsvZerleger zerleger) : IFunktion
{
    public async IAsyncEnumerable<OneOf<Zeile, Fortschritt, Bilanz>> Auf(KatalogBereit k)
    {
        var nummer = 0;
        var fehler = 0;
        await foreach (var text in File.ReadLinesAsync(k.Pfad))
        {
            nummer++;
            var felder = zerleger.Zerlege(text);
            if (felder.Length < 3) fehler++;
            yield return new Zeile(k.Import, nummer, felder);
            if (nummer % 100 == 0) yield return new Fortschritt(k.Import, nummer);
        }
        yield return new Bilanz(k.Import, nummer, fehler);
    }
}

// Im Unterlauf je Zeile: der Artikel (Id aus der Herkunft und der Artikelnummer).

}

// Die Prognose: der Kopf bildet eine Tendenz, die GABELT an zwei Knoten DERSELBEN GENERISCHEN Funktion (die Glättung, je Typargument ein Baustein:
// kurzes und langes Fenster, jede mit eigenem Konfig-Record am Knoten); das Prognosemodell FÜHRT beide in seinem ZUSTAND ZUSAMMEN (IZustand: je
// Glättung ein Auf, die erste wartet im Zustand auf die zweite, bis dahin ein leerer Strom), nimmt eine Technik mit eigenem Zustand und gibt in einer
// WAHL entweder die Prognose oder eine Unwetter-Warnung als HINWEIS aus.
//
// Die Kartenausgabe (AUSDRÜCKLICHE KANTEN, docs/04 §4.4): sie zeichnet eine Karte und filtert sie in einem Graphen im Graphen (Kantenglättung,
// Schärfung, nochmals Kantenglättung — DIESELBE Funktion zweimal, eine Kette Karte → Karte → Karte, kein Kreis); den Ausgang des Filters nehmen
// zwei Knoten DERSELBEN Skalierung (Vorschau und Daumenbild, je eigene Konfig) — das Daumenbild VERLIERBAR; die Ablage trägt [Kopf] UND [Von].

public sealed class Tendenzbildung : IFunktion
{
    public Tendenz Auf(Bilde b) => new(b.Station, b.Grad);
}

    public Dictionary<Guid, int> Anzahl { get; set; } = new();
    public Dictionary<Guid, double> Kurz { get; set; } = new();

// Technik (ohne Marker, im Konstruktor des Modells) mit eigenem Zustand: zählt die Prognosen — ihr Zustand gehört dem Knoten des Modells.
public interface IStatistik
{
    void Zaehle(Guid station);
}

public sealed class Zaehlung
{
    public Dictionary<Guid, int> Je { get; set; } = new();
}

public sealed class Statistik(IZustand<Zaehlung> zaehlung) : IStatistik
{
    public void Zaehle(Guid station) => zaehlung.Wert.Je[station] = zaehlung.Wert.Je.GetValueOrDefault(station) + 1;
}

// ZUSAMMENFÜHREN im Logik-Code: je Glättung ein Auf; der Zustand hält die erste, bis die zweite derselben Station da ist (bis dahin ein leerer
// Strom); dann mit Technik eine WAHL zwischen Prognose und Warnung (Hinweis).
public sealed class Prognosemodell(IZustand<Gewichte> gewichte, IStatistik statistik) : IFunktion
{
    public IEnumerable<OneOf<Prognostiziert, Unwetter>> Auf(Geglaettet<Kurzfenster> kurz)
    {
        gewichte.Wert.Kurz[kurz.Station] = kurz.Grad;
        return Prognose(kurz.Station);
    }

    public IEnumerable<OneOf<Prognostiziert, Unwetter>> Auf(Geglaettet<Langfenster> lang)
    {
        gewichte.Wert.Lang[lang.Station] = lang.Grad;
        return Prognose(lang.Station);
    }

    private IEnumerable<OneOf<Prognostiziert, Unwetter>> Prognose(Guid s)
    {
        if (!gewichte.Wert.Kurz.TryGetValue(s, out var kurz) || !gewichte.Wert.Lang.TryGetValue(s, out var lang)) yield break;
        gewichte.Wert.Kurz.Remove(s);
        gewichte.Wert.Lang.Remove(s);
        statistik.Zaehle(s);
        var n = gewichte.Wert.Anzahl.GetValueOrDefault(s) + 1;
        gewichte.Wert.Anzahl[s] = n;
        gewichte.Wert.Mittel[s] = gewichte.Wert.Mittel.GetValueOrDefault(s) + (kurz - gewichte.Wert.Mittel.GetValueOrDefault(s)) / n;
        if (Math.Abs(kurz - lang) > 10) yield return new Unwetter(s, "rot");
        else yield return new Prognostiziert(s, gewichte.Wert.Mittel[s], n);
    }
}
