using System.Runtime.CompilerServices;
using Abstractions;

namespace Sondenkatalog.Katalog.Messung;

// ── Die Quelle: eine eigene Quellfunktion (Lausche) mit Konfig, gesichertem Zustand und Technik ─────────────────────────────────────────────

/// <summary>Der Zustand der Abfrage: wie oft sie gelesen hat (gesichert mit dem Lauf; nach einem Neustart zählt sie weiter).</summary>
public sealed class Abfragestand
{
    public long Abfragen { get; set; }
}

public interface ISensorbus
{
    Task<double?> LiesAsync(Guid sensor, CancellationToken ct);
}

public sealed class Feldbus : ISensorbus
{
    public Task<double?> LiesAsync(Guid sensor, CancellationToken ct) => Task.FromResult<double?>(null);
}

public sealed class Sensorabfrage(SensorabfrageKonfig konfig, IZustand<Abfragestand> stand, ISensorbus bus) : IFunktion
{
    public async IAsyncEnumerable<Rohwert> Lausche([EnumeratorCancellation] CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            foreach (var s in konfig.Sensoren)
                if (await bus.LiesAsync(Guid.Parse(s), ct) is { } wert)
                {
                    stand.Wert.Abfragen++;
                    yield return new Rohwert(Guid.Parse(s), wert, DateTimeOffset.UtcNow);
                }
            await Task.Delay(konfig.Intervall, ct);
        }
    }
}

// Die Aufbereitung: ein unlesbarer Wert ergibt nichts (ein leerer Strom), ein lesbarer den geprüften Wert.
public sealed class Plausibilisierung : IFunktion
{
    public async IAsyncEnumerable<Plausibel> Auf(Rohwert r)
    {
        if (!double.IsNaN(r.Wert) && !double.IsInfinity(r.Wert)) yield return new Plausibel(r.Sensor, r.Wert, r.GemessenAm);
    }
}

public sealed class Messwertpruefung : IFunktion
{
    public async Task<Geprueft> Auf(Plausibel p) => new(p.Sensor, p.Wert, p.GemessenAm);
}

// Der Alarm öffnet die Störung desselben Sensors (dieselbe Id wie der Sensor).
public sealed class StoerungBeiAlarm : IFunktion
{
    public async Task<Stoerung> Auf(Alarm a) => new(a.Sensor, a.Wert);
}

// ── RENNEN: die Bereitschaft wird über zwei Wege gesucht; jeder Weg meldet für sich ──────────────────────────────────────────────────────────
public sealed class Alarmverteilung : IFunktion
{
    public async Task<AlarmVerteilt> Auf(Alarmiere a) => new(a.Fall);
}

public interface ITelefonie
{
    Task<string> RufeAsync(Guid fall);
}

public sealed class Telefonanlage : ITelefonie
{
    public Task<string> RufeAsync(Guid fall) => Task.FromResult("110");
}

public sealed class Anrufkette(ITelefonie telefonie) : IFunktion
{
    public async Task<PerAnrufErreicht> Auf(AlarmVerteilt a) => new(a.Fall, await telefonie.RufeAsync(a.Fall));
}

public sealed class Funkruf : IFunktion
{
    public async Task<PerFunkErreicht> Auf(AlarmVerteilt a, CancellationToken ct)
    {
        await Task.Delay(TimeSpan.FromSeconds(5), ct);
        return new PerFunkErreicht(a.Fall, "Kanal 2");
    }
}

public sealed class Alarmquittung : IFunktion
{
    public async Task<Erreicht> Auf(PerAnrufErreicht anruf) => new(anruf.Fall, "Anruf " + anruf.Nummer);

    public async Task<Erreicht> Auf(PerFunkErreicht funk) => new(funk.Fall, "Funk " + funk.Kanal);
}
