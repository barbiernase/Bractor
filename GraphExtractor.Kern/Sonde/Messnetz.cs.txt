using System.Runtime.CompilerServices;
using Abstractions;

// SONDE — fremdes Szenario 2 (docs/12 §12.6): SENSOR-STROM MIT ALARM.
//
// Viele Messwerte, wenige Fakten: ein Sensor zählt Überschreitungen seines Grenzwerts hintereinander (Zähl-Aggregat) und löst nach N den
// Alarm aus; ein Messwert im Rahmen ändert nichts. Die Werte holt eine EIGENE QUELLFUNKTION (Lausche statt Auf, Entscheidung 8) am Kopf eines
// Graphen — mit Konfig, gesichertem Zustand und Technik; dahinter liegt ein GRAPH IM GRAPHEN (die Aufbereitung, eine Schachtel mit Signatur).
// Der Alarm öffnet die Störung desselben Sensors über einen Graphen am Kopf (Command ohne Ziel aus dem Decide). Die geöffnete Störung gibt ARBEIT
// an die Schichtleitung (ein Akteur hört einen Command ohne Ziel aus dem Decide und antwortet mit seinem IDarf) und alarmiert die Bereitschaft:
// ein RENNEN zweier Wege — der erste gewinnt, der andere wird abgebrochen.

namespace Anlagen.Messung;

public record RichteSensorEin(Id<Sensor> AggregateId, string Ort, double Grenzwert, int AlarmNach) : ICommand<Sensor>, ICreationCommand;
public record SensorEingerichtet(string Ort, double Grenzwert, int AlarmNach) : IEvent;
public record SensorUnbekannt(Id<Sensor> SensorId) : ITransientEvent;

public record MeldeMesswert(Id<Sensor> AggregateId, double Wert, DateTimeOffset GemessenAm) : ICommand<Sensor>;
public record GrenzwertUeberschritten(double Wert, DateTimeOffset GemessenAm, int Hintereinander) : IEvent;
public record AlarmAusgeloest(double Wert, DateTimeOffset GemessenAm) : IEvent;
public record WertWiederImRahmen(double Wert, DateTimeOffset GemessenAm) : IEvent;

/// <summary>Ein Command ohne Ziel aus dem Decide des Sensors: die Störung desselben Sensors ist zu öffnen.</summary>
public record OeffneStoerungFuer(Id<Sensor> Sensor, double Wert) : ICommand;

public partial class Sensor : IState
{
    public double Grenzwert { get; set; }
    public int AlarmNach { get; set; }
    public int Hintereinander { get; set; }
    public bool Alarm { get; set; }

    public partial class Decider : IDecider<Sensor>
    {
        public IEnumerable<OneOf<SensorEingerichtet>> Decide(RichteSensorEin c)
        {
            if (this.State.Version > 0) yield break;
            yield return new SensorEingerichtet(c.Ort, c.Grenzwert, c.AlarmNach);
        }

        public IEnumerable<OneOf<GrenzwertUeberschritten, AlarmAusgeloest, WertWiederImRahmen, OeffneStoerungFuer, SensorUnbekannt>> Decide(MeldeMesswert c)
        {
            if (this.State.Version == 0) { yield return new SensorUnbekannt(c.AggregateId); yield break; }
            if (c.Wert <= this.State.Grenzwert)
            {
                if (this.State.Hintereinander > 0) yield return new WertWiederImRahmen(c.Wert, c.GemessenAm);
                yield break;
            }
            yield return new GrenzwertUeberschritten(c.Wert, c.GemessenAm, this.State.Hintereinander + 1);
            if (!this.State.Alarm && this.State.Hintereinander + 1 >= this.State.AlarmNach)
            {
                yield return new AlarmAusgeloest(c.Wert, c.GemessenAm);
                yield return new OeffneStoerungFuer(c.AggregateId, c.Wert);
            }
        }
    }

    public partial class Applier : IApplier<Sensor>
    {
        public void Apply(SensorEingerichtet e) { this.State.Grenzwert = e.Grenzwert; this.State.AlarmNach = e.AlarmNach; }
        public void Apply(GrenzwertUeberschritten e) { this.State.Hintereinander = e.Hintereinander; }
        public void Apply(AlarmAusgeloest e) { this.State.Alarm = true; }
        public void Apply(WertWiederImRahmen e) { this.State.Hintereinander = 0; this.State.Alarm = false; }
    }
}

public record OeffneStoerung(Id<Stoerungsfall> AggregateId, double Wert) : ICommand<Stoerungsfall>;
public record StoerungGeoeffnet(double Wert) : IEvent;
public record StoerungLaeuftSchon(Id<Stoerungsfall> SensorId) : ITransientEvent;

public record QuittiereStoerung(Id<Stoerungsfall> AggregateId, string Von) : ICommand<Stoerungsfall>;
public record StoerungQuittiert(string Von) : IEvent;
public record KeineStoerung(Id<Stoerungsfall> SensorId) : ITransientEvent;

public record VermerkeAlarmierung(Id<Stoerungsfall> AggregateId, string Weg) : ICommand<Stoerungsfall>;
public record AlarmierungVermerkt(string Weg) : IEvent;

/// <summary>Commands ohne Ziel aus dem Decide der Störung: Arbeit für die Schichtleitung und der Kopf der Alarmierung.</summary>
public record PruefeStoerungVorOrt(Id<Stoerungsfall> Fall, double Wert) : ICommand;
public record AlarmiereBereitschaft(Id<Stoerungsfall> Fall) : ICommand;

// Je Sensor ein Störungsfall (dieselbe Id wie der Sensor): offen oder nicht — eine Störung läuft höchstens einmal zugleich.
public partial class Stoerungsfall : IState
{
    public bool Offen { get; set; }

    public partial class Decider : IDecider<Stoerungsfall>
    {
        public IEnumerable<OneOf<StoerungGeoeffnet, PruefeStoerungVorOrt, AlarmiereBereitschaft, StoerungLaeuftSchon>> Decide(OeffneStoerung c)
        {
            if (this.State.Offen) { yield return new StoerungLaeuftSchon(c.AggregateId); yield break; }
            yield return new StoerungGeoeffnet(c.Wert);
            yield return new PruefeStoerungVorOrt(c.AggregateId, c.Wert);
            yield return new AlarmiereBereitschaft(c.AggregateId);
        }

        public IEnumerable<OneOf<StoerungQuittiert, KeineStoerung>> Decide(QuittiereStoerung c)
        {
            if (!this.State.Offen) { yield return new KeineStoerung(c.AggregateId); yield break; }
            yield return new StoerungQuittiert(c.Von);
        }

        public IEnumerable<OneOf<AlarmierungVermerkt>> Decide(VermerkeAlarmierung c)
        {
            yield return new AlarmierungVermerkt(c.Weg);
        }
    }

    public partial class Applier : IApplier<Stoerungsfall>
    {
        public void Apply(StoerungGeoeffnet e) { this.State.Offen = true; }
        public void Apply(StoerungQuittiert e) { this.State.Offen = false; }
        public void Apply(AlarmierungVermerkt e) { }
    }
}

// ── Die Quelle: eine eigene Quellfunktion (Lausche) mit Konfig, gesichertem Zustand und Technik ─────────────────────────────────────────────
public record Rohwert(Guid Sensor, double Wert, DateTimeOffset GemessenAm) : ICommand;
public record Plausibel(Guid Sensor, double Wert, DateTimeOffset GemessenAm) : ICommand;

public record SensorabfrageKonfig(string[] Sensoren, TimeSpan Intervall);

/// <summary>Der Zustand der Abfrage: wie oft sie gelesen hat (gesichert mit dem Lauf; nach einem Neustart zählt sie weiter).</summary>
public sealed class Abfragestand
{
    public long Abfragen { get; set; }
}

public interface ISensorbus
{
    Task<double?> LiesAsync(Guid sensor, CancellationToken ct);
}

public sealed class Feldbus : ISensorbus
{
    public Task<double?> LiesAsync(Guid sensor, CancellationToken ct) => Task.FromResult<double?>(null);
}

public sealed class Sensorabfrage(SensorabfrageKonfig konfig, IZustand<Abfragestand> stand, ISensorbus bus) : IFunktion
{
    public async IAsyncEnumerable<Rohwert> Lausche([EnumeratorCancellation] CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            foreach (var s in konfig.Sensoren)
                if (await bus.LiesAsync(Guid.Parse(s), ct) is { } wert)
                {
                    stand.Wert.Abfragen++;
                    yield return new Rohwert(Guid.Parse(s), wert, DateTimeOffset.UtcNow);
                }
            await Task.Delay(konfig.Intervall, ct);
        }
    }
}

// Die Aufbereitung: ein unlesbarer Wert ergibt nichts (ein leerer Strom), ein lesbarer den Command an den Sensor.
public sealed class Plausibilisierung : IFunktion
{
    public IEnumerable<Plausibel> Auf(Rohwert r)
    {
        if (!double.IsNaN(r.Wert) && !double.IsInfinity(r.Wert)) yield return new Plausibel(r.Sensor, r.Wert, r.GemessenAm);
    }
}

public sealed class Messwertpruefung : IFunktion
{
    public MeldeMesswert Auf(Plausibel p) => new(new Id<Sensor>(p.Sensor), p.Wert, p.GemessenAm);
}

// GRAPH IM GRAPHEN: die Aufbereitung ist eine Schachtel (Signatur Rohwert → MeldeMesswert), die Messkette setzt die Quelle davor.
public sealed record Messwertaufbereitung([Kopf] Plausibilisierung Plausi, [Von(nameof(Plausi))] Messwertpruefung Pruefung) : IGraph;
public sealed record Messkette(Sensorabfrage Abfrage, [Von(nameof(Abfrage))] Messwertaufbereitung Aufbereitung) : IGraph;

// Der Alarm öffnet die Störung desselben Sensors (dieselbe Id wie der Sensor).
public sealed class StoerungBeiAlarm : IFunktion
{
    public OeffneStoerung Auf(OeffneStoerungFuer a) => new(new Id<Stoerungsfall>(a.Sensor.Wert), a.Wert);
}

public sealed record Stoerungsoeffnung([Kopf] StoerungBeiAlarm Oeffnung) : IGraph;

// ── RENNEN: die Bereitschaft wird über zwei Wege gesucht; wer zuerst erreicht ist, gewinnt (der andere Zweig wird abgebrochen) ──────────────
public record AlarmVerteilt(Id<Stoerungsfall> Fall) : ICommand;
public record PerAnrufErreicht(Id<Stoerungsfall> Fall, string Nummer) : ICommand;
public record PerFunkErreicht(Id<Stoerungsfall> Fall, string Kanal) : ICommand;

public sealed class Alarmverteilung : IFunktion
{
    public AlarmVerteilt Auf(AlarmiereBereitschaft a) => new(a.Fall);
}

public interface ITelefonie
{
    Task<string> RufeAsync(Guid fall);
}

public sealed class Telefonanlage : ITelefonie
{
    public Task<string> RufeAsync(Guid fall) => Task.FromResult("110");
}

public sealed class Anrufkette(ITelefonie telefonie) : IFunktion
{
    public async Task<PerAnrufErreicht> Auf(AlarmVerteilt a) => new(a.Fall, await telefonie.RufeAsync(a.Fall.Wert));
}

public sealed class Funkruf : IFunktion
{
    public async Task<PerFunkErreicht> Auf(AlarmVerteilt a, CancellationToken ct)
    {
        await Task.Delay(TimeSpan.FromSeconds(5), ct);
        return new PerFunkErreicht(a.Fall, "Kanal 2");
    }
}

public sealed class Alarmquittung : IFunktion
{
    public VermerkeAlarmierung Auf(OneOf<PerAnrufErreicht, PerFunkErreicht> erster) => erster.Value switch
    {
        PerAnrufErreicht anruf => new VermerkeAlarmierung(anruf.Fall, "Anruf " + anruf.Nummer),
        PerFunkErreicht funk => new VermerkeAlarmierung(funk.Fall, "Funk " + funk.Kanal),
        _ => throw new InvalidOperationException("kein Fall"),
    };
}

public sealed record Alarmierung(
    [Kopf] Alarmverteilung Verteilung,
    [Von(nameof(Verteilung))] Anrufkette Anruf,
    [Von(nameof(Verteilung))] Funkruf Funk,
    [Von(nameof(Anruf), nameof(Funk))] Alarmquittung Quittung) : IGraph;

// Wer hineingibt: das Sensornetz (Akteur der Instanz der Messkette — es darf, was sie verlässt); die Schichtleitung richtet ein, nimmt
// ARBEIT an (die Prüfung vor Ort) und antwortet mit der Quittung; sie hört Alarm und Störung (Kenntnis).
public sealed record Sensornetz : IAkteur, IDarf<MeldeMesswert>;
public sealed record Schichtleitung : IAkteur, IDarf<RichteSensorEin>, IDarf<QuittiereStoerung>,
    IHoert<PruefeStoerungVorOrt>, IHoert<AlarmAusgeloest>, IHoert<StoerungGeoeffnet>;

public sealed record SensorGateway : IClient, IVerkoerpert<Sensornetz>;
public sealed record Leitstand : IClient, IVerkoerpert<Schichtleitung>;
