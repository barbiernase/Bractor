using Abstractions;
using Sondenkatalog.Katalog;

// SONDE — die Übersetzungen des Messnetzes (docs/04 §4.4): der geprüfte Messwert an den Sensor, der Alarm der Schichtleitung in den
// Kopf der Störungsöffnung und deren Ergebnis an den Störungsfall, die Alarmierung hinein und jede Erreichung an den Fall (Guid ⇄ Id ausdrücklich).

namespace Anlagen.Messung;

#pragma warning disable CS9113

public sealed class Geprueft_MeldeMesswert_1(Geprueft source) : ITranslate<Geprueft, MeldeMesswert>
{
    public MeldeMesswert Translate() => new(AggregateId: new(source.Sensor), Wert: source.Wert, GemessenAm: source.GemessenAm);
}

public sealed class OeffneStoerungFuer_Alarm_1(OeffneStoerungFuer source) : ITranslate<OeffneStoerungFuer, Alarm>
{
    public Alarm Translate() => new(Sensor: source.Sensor.Wert, Wert: source.Wert);
}

public sealed class Stoerung_OeffneStoerung_1(Stoerung source) : ITranslate<Stoerung, OeffneStoerung>
{
    public OeffneStoerung Translate() => new(AggregateId: new(source.Fall), Wert: source.Wert);
}

public sealed class AlarmiereBereitschaft_Alarmiere_1(AlarmiereBereitschaft source) : ITranslate<AlarmiereBereitschaft, Alarmiere>
{
    public Alarmiere Translate() => new(Fall: source.Fall.Wert);
}

public sealed class Erreicht_VermerkeAlarmierung_1(Erreicht source) : ITranslate<Erreicht, VermerkeAlarmierung>
{
    public VermerkeAlarmierung Translate() => new(AggregateId: new(source.Fall), Weg: source.Weg);
}
