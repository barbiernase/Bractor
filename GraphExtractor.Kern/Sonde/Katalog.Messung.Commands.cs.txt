using Abstractions;

// SONDE — fremder Katalog, Bereich Messung (docs/04 §4.4): Messkette, Störungsöffnung und Alarmierung domänenneutral — Sensor und
// Fall sind Guids.

namespace Sondenkatalog.Katalog.Messung;

// Die Messkette: der rohe Wert aus der Quelle, der plausible, der geprüfte (ihr Ergebnis).
public record Rohwert(Guid Sensor, double Wert, DateTimeOffset GemessenAm) : ICommand;
public record Plausibel(Guid Sensor, double Wert, DateTimeOffset GemessenAm) : ICommand;
public record Geprueft(Guid Sensor, double Wert, DateTimeOffset GemessenAm) : ICommand;

// Die Störungsöffnung: aus dem Alarm eines Sensors die Störung desselben Falls (dieselbe Id).
public record Alarm(Guid Sensor, double Wert) : ICommand;
public record Stoerung(Guid Fall, double Wert) : ICommand;

// RENNEN: die Bereitschaft wird über zwei Wege gesucht — jeder meldet für sich.
public record Alarmiere(Guid Fall) : ICommand;
public record AlarmVerteilt(Guid Fall) : ICommand;
public record PerAnrufErreicht(Guid Fall, string Nummer) : ICommand;
public record PerFunkErreicht(Guid Fall, string Kanal) : ICommand;
public record Erreicht(Guid Fall, string Weg) : ICommand;
