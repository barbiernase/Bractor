using Abstractions;
using Leihwesen.Ausleihe;
using Leihwesen.Gebuehren;
using Leihwesen.Lesen;

// SONDE — ein ABLAUF ohne Fluss-Sprache (docs/10 §10.3; früher ein Fluss mit Je/Sammle, ∧/∨, ⏳ und ✕): Vorgangs-Aggregat + Funktionen + Frist.
//         · eine eigene Quell-Nachricht (IQuellNachricht, Kennung vom Client) und eine Quelle mit fest codiertem Treiber (DateiGemeldet);
//         · der Start als gleichbleibende Funktion (neue Id aus der Herkunft, R13);
//         · der Vorgang als ZÄHL-AGGREGAT (keine Collection im Zustand): Fan-out über eine Funktion draußen, die als STROM je Seite einen
//           Command liefert, Fan-in durch Zählen im Aggregat; das Rennen gegen die Zeit ist eine FRIST desselben Aggregats;
//         · eine Funktion mit LESE-Fähigkeit (läuft im Host, hält ihre Ausgabe fest, R10/R12) auf dem Event einer Frist (Ausleihe).

namespace Leihwesen.Pruefung;

public sealed record RueckgabeGescannt(Guid LeserId, string Code, IReadOnlyList<string> Seiten) : IQuellNachricht
{
    public string Kennung => Code;
}

// Eine Quelle mit fest codiertem Treiber (docs/04 §4.7): nur der Typ, der Datei-Treiber befüllt die geerbten Felder.
public sealed record AusweisAbgelegt : DateiGemeldet;

public record StartePruefung(Id<Seitenpruefung> AggregateId, Guid LeserId, IReadOnlyList<string> Seiten) : ICommand<Seitenpruefung>, ICreationCommand;
public record PruefungGestartet(Guid LeserId, IReadOnlyList<string> Seiten) : IEvent;
public record PruefungLaeuftSchon(Id<Seitenpruefung> PruefungId) : ITransientEvent;

public record ZaehleSeite(Id<Seitenpruefung> AggregateId, int Nummer, bool Beschaedigt) : ICommand<Seitenpruefung>;
public record SeiteGezaehlt(int Nummer, bool Beschaedigt) : IEvent;
public record PruefungAbgeschlossen(Guid LeserId, int Beschaedigte) : IEvent;

public record BrichPruefungAb(Id<Seitenpruefung> AggregateId) : ICommand<Seitenpruefung>;
public record PruefungAbgebrochen(int Gezaehlt) : IEvent;

public partial class Seitenpruefung : IState
{
    public Guid LeserId { get; set; }
    public int Seiten { get; set; }
    public int Gezaehlt { get; set; }
    public int Beschaedigte { get; set; }
    public bool Fertig { get; set; }

    public partial class Decider : IDecider<Seitenpruefung>
    {
        public IEnumerable<OneOf<PruefungGestartet, PruefungLaeuftSchon, Frist<BrichPruefungAb>>> Decide(StartePruefung c)
        {
            if (this.State.Version > 0) { yield return new PruefungLaeuftSchon(c.AggregateId); yield break; }
            yield return new PruefungGestartet(c.LeserId, c.Seiten);
            yield return new Frist<BrichPruefungAb>(new BrichPruefungAb(c.AggregateId), TimeSpan.FromSeconds(30));
        }

        public IEnumerable<OneOf<SeiteGezaehlt, PruefungAbgeschlossen>> Decide(ZaehleSeite c)
        {
            if (this.State.Fertig) yield break;
            yield return new SeiteGezaehlt(c.Nummer, c.Beschaedigt);
            if (this.State.Gezaehlt + 1 == this.State.Seiten)
                yield return new PruefungAbgeschlossen(this.State.LeserId, this.State.Beschaedigte + (c.Beschaedigt ? 1 : 0));
        }

        public IEnumerable<OneOf<PruefungAbgebrochen>> Decide(BrichPruefungAb c)
        {
            if (this.State.Fertig) yield break;
            yield return new PruefungAbgebrochen(this.State.Gezaehlt);
        }
    }

    public partial class Applier : IApplier<Seitenpruefung>
    {
        public void Apply(PruefungGestartet e) { this.State.LeserId = e.LeserId; this.State.Seiten = e.Seiten.Count; }
        public void Apply(SeiteGezaehlt e) { this.State.Gezaehlt++; if (e.Beschaedigt) this.State.Beschaedigte++; }
        public void Apply(PruefungAbgeschlossen e) { this.State.Fertig = true; }
        public void Apply(PruefungAbgebrochen e) { this.State.Fertig = true; }
    }
}

// Der Start: je gescannter Rückgabe ein Vorgang.
public sealed class Pruefungsstart : IFunktion
{
    public StartePruefung Auf(RueckgabeGescannt e, Herkunft h) => new(h.NeueId<Seitenpruefung>(), e.LeserId, e.Seiten);
}

// Fan-out draußen: je Seite ein Command an den Vorgang (Strom; nach einem Absturz geht es bei der nächsten Seite weiter).
public interface ISeitenPruefung : IFunktion
{
    IAsyncEnumerable<ZaehleSeite> Auf(PruefungGestartet e, Herkunft h);
}

// Fan-in: ist der Vorgang mit beschädigten Seiten abgeschlossen, wird ein Schadensfall gemeldet.
public sealed class SchadenAusPruefung : IFunktion
{
    public MeldeSchaden? Auf(PruefungAbgeschlossen e, Herkunft h) =>
        e.Beschaedigte > 0 ? new MeldeSchaden(h.NeueId<Schadensfall>(), e.LeserId) : null;
}

// Lese-Fähigkeit: die Mahnung nach Ablauf der Leihfrist (Ausleihe) richtet sich nach der Zahl offener Ausleihen des Lesers.
public sealed class Offenzaehlung : IFunktion
{
    public async Task<EroeffneMahnung?> Auf(RueckgabeAngemahnt e, Herkunft h, IZaehleOffene offene)
    {
        var anzahl = (await offene.OffeneAsync(e.LeserId)).Count;
        return anzahl == 0 ? null : new EroeffneMahnung(h.NeueId<Mahnvorgang>(), anzahl * 2m);
    }
}
