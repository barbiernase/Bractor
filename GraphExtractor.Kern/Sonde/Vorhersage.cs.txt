using Abstractions;

// SONDE — fremdes Szenario (docs/04 §4.4): GENERISCHE FUNKTIONEN, ZUSTAND UND HINWEISE.
//
// Wetterstationen melden Temperaturen (Fakten im Log); das Decide gibt je Meldung einen Command ohne Ziel an den Kopf der Prognose. Der Kopf bildet
// eine Tendenz, die GABELT an zwei Knoten DERSELBEN GENERISCHEN Funktion (die Glättung, je Typargument ein Baustein: kurzes und langes Fenster,
// jede mit eigenem Konfig-Record am Knoten); das Prognosemodell FÜHRT beide in seinem ZUSTAND ZUSAMMEN (IZustand: je Glättung ein Auf, die erste
// wartet im Zustand auf die zweite, bis dahin ein leerer Strom), nimmt
// eine Technik mit eigenem Zustand und gibt in einer WAHL entweder die Prognose als Command an die Station oder eine Unwetter-Warnung als HINWEIS
// aus — Hinweise gehen nur an Akteure draußen (der Warndienst hört sie), keine Funktion hört einen.
//
// Dazu AUSDRÜCKLICHE KANTEN (docs/04 §4.4): die Kartenausgabe zeichnet eine Karte und filtert sie in einem Graphen im Graphen (Kantenglättung,
// Schärfung, nochmals Kantenglättung — DIESELBE Funktion zweimal, eine Kette Karte → Karte → Karte, kein Kreis); den Ausgang des Filters nehmen
// zwei Knoten DERSELBEN Skalierung (Vorschau und Daumenbild, je eigene Konfig); die Ablage trägt [Kopf] UND [Von]: sie nimmt die Vorschau von innen
// und das Löschen von außen.

namespace Wetterdienst.Vorhersage;

public record MeldeTemperatur(Id<Station> AggregateId, double Grad) : ICommand<Station>;
public record TemperaturGemeldet(double Grad) : IEvent;
public record StationStillgelegt(Id<Station> StationId) : ITransientEvent;

public record VermerkePrognose(Id<Station> AggregateId, double Grad, int Gelernt) : ICommand<Station>;
public record PrognoseVermerkt(double Grad, int Gelernt) : IEvent;

/// <summary>Ein Command ohne Ziel aus dem Decide: aus der neuen Temperatur ist eine Prognose zu bilden.</summary>
public record BildePrognose(Id<Station> Station, double Grad) : ICommand;

public partial class Station : IState
{
    public bool Stillgelegt { get; set; }
    public double Letzte { get; set; }

    public partial class Decider : IDecider<Station>
    {
        public IEnumerable<OneOf<TemperaturGemeldet, BildePrognose, StationStillgelegt>> Decide(MeldeTemperatur c)
        {
            if (this.State.Stillgelegt) { yield return new StationStillgelegt(c.AggregateId); yield break; }
            yield return new TemperaturGemeldet(c.Grad);
            yield return new BildePrognose(c.AggregateId, c.Grad);
        }

        public IEnumerable<OneOf<PrognoseVermerkt>> Decide(VermerkePrognose c)
        {
            yield return new PrognoseVermerkt(c.Grad, c.Gelernt);
        }
    }

    public partial class Applier : IApplier<Station>
    {
        public void Apply(TemperaturGemeldet e) { this.State.Letzte = e.Grad; }
        public void Apply(PrognoseVermerkt e) { }
    }
}

// Was nur Knoten einander geben — auch generisch (das Typargument unterscheidet die beiden Glättungen).
public record Tendenz(Id<Station> Station, double Grad) : ICommand;
public record Geglaettet<TFenster>(Id<Station> Station, double Grad) : ICommand where TFenster : IFenster;

public interface IFenster { }
public sealed record Kurzfenster : IFenster;
public sealed record Langfenster : IFenster;

public sealed class Tendenzbildung : IFunktion
{
    public Tendenz Auf(BildePrognose b) => new(b.Station, b.Grad);
}

// Eine GENERISCHE Funktion: die offene Definition ist eine Blaupause, je Typargument eines Knotens ein Baustein.
public record GlaettungsKonfig(double Gewicht);

public sealed class Glaettung<TFenster>(GlaettungsKonfig konfig) : IFunktion where TFenster : IFenster
{
    public Geglaettet<TFenster> Auf(Tendenz t) => new(t.Station, t.Grad * konfig.Gewicht);
}

// Der Zustand des Modells: je Station, was es gelernt hat (kein Domänen-Zustand; gesichert, wenn der Lauf fertig ist).
public sealed class Gewichte
{
    public Dictionary<Guid, double> Mittel { get; set; } = new();
    public Dictionary<Guid, int> Anzahl { get; set; } = new();
    public Dictionary<Guid, double> Kurz { get; set; } = new();
    public Dictionary<Guid, double> Lang { get; set; } = new();
}

// Technik (ohne Marker, im Konstruktor des Modells) mit eigenem Zustand: zählt die Prognosen — ihr Zustand gehört dem Knoten des Modells.
public interface IStatistik
{
    void Zaehle(Guid station);
}

public sealed class Zaehlung
{
    public Dictionary<Guid, int> Je { get; set; } = new();
}

public sealed class Statistik(IZustand<Zaehlung> zaehlung) : IStatistik
{
    public void Zaehle(Guid station) => zaehlung.Wert.Je[station] = zaehlung.Wert.Je.GetValueOrDefault(station) + 1;
}

public record Unwetter(Id<Station> Station, string Stufe) : ITransientEvent;

// ZUSAMMENFÜHREN im Logik-Code: je Glättung ein Auf; der Zustand hält die erste, bis die zweite derselben Station da ist (bis dahin ein leerer
// Strom); dann mit Technik eine WAHL zwischen Prognose (Command an die Station) und Warnung (Hinweis).
public sealed class Prognosemodell(IZustand<Gewichte> gewichte, IStatistik statistik) : IFunktion
{
    public IEnumerable<OneOf<VermerkePrognose, Unwetter>> Auf(Geglaettet<Kurzfenster> kurz)
    {
        gewichte.Wert.Kurz[kurz.Station.Wert] = kurz.Grad;
        return Prognose(kurz.Station);
    }

    public IEnumerable<OneOf<VermerkePrognose, Unwetter>> Auf(Geglaettet<Langfenster> lang)
    {
        gewichte.Wert.Lang[lang.Station.Wert] = lang.Grad;
        return Prognose(lang.Station);
    }

    private IEnumerable<OneOf<VermerkePrognose, Unwetter>> Prognose(Id<Station> station)
    {
        var s = station.Wert;
        if (!gewichte.Wert.Kurz.TryGetValue(s, out var kurz) || !gewichte.Wert.Lang.TryGetValue(s, out var lang)) yield break;
        gewichte.Wert.Kurz.Remove(s);
        gewichte.Wert.Lang.Remove(s);
        statistik.Zaehle(s);
        var n = gewichte.Wert.Anzahl.GetValueOrDefault(s) + 1;
        gewichte.Wert.Anzahl[s] = n;
        gewichte.Wert.Mittel[s] = gewichte.Wert.Mittel.GetValueOrDefault(s) + (kurz - gewichte.Wert.Mittel.GetValueOrDefault(s)) / n;
        if (Math.Abs(kurz - lang) > 10) yield return new Unwetter(station, "rot");
        else yield return new VermerkePrognose(station, gewichte.Wert.Mittel[s], n);
    }
}

public sealed record Prognose(
    [Kopf] Tendenzbildung Tendenz,
    [Von(nameof(Tendenz))] Glaettung<Kurzfenster> Kurz,
    [Von(nameof(Tendenz))] Glaettung<Langfenster> Lang,
    [Von(nameof(Kurz), nameof(Lang))] Prognosemodell Modell) : IGraph;

// ── Kartenausgabe: jede Kante steht am nehmenden Knoten ──
/// <summary>Ein Command ohne Ziel vom Warndienst an den Kopf der Kartenausgabe: die Karte einer Station ist zu zeichnen.</summary>
public record ZeichneKarte(Id<Station> Station, double Grad) : ICommand;
/// <summary>Ein Command ohne Ziel vom Warndienst an die Ablage (ein Kopf, der auch von innen nimmt): die Karte ist zu löschen.</summary>
public record LoescheKarte(Id<Station> Station) : ICommand;
public record KarteGezeichnet(Id<Station> Station, int Breite) : ICommand;
public record KarteSkaliert(Id<Station> Station, int Breite) : ICommand;
public record KarteBereit(Id<Station> Station, int Breite) : ITransientEvent;
public record KarteGeloescht(Id<Station> Station) : ITransientEvent;

public sealed class Kartenzeichnung : IFunktion
{
    public KarteGezeichnet Auf(ZeichneKarte z) => new(z.Station, 2048);
}

public sealed class Kantenglaettung : IFunktion
{
    public KarteGezeichnet Auf(KarteGezeichnet k) => k;
}

public sealed class Kartenschaerfung : IFunktion
{
    public KarteGezeichnet Auf(KarteGezeichnet k) => k;
}

public record SkalierungsKonfig(int Breite);

public sealed class Kartenskalierung(SkalierungsKonfig konfig) : IFunktion
{
    public KarteSkaliert Auf(KarteGezeichnet k) => new(k.Station, konfig.Breite);
}

public sealed class Kartenablage : IFunktion
{
    public KarteBereit Auf(KarteSkaliert k) => new(k.Station, k.Breite);
    public KarteGeloescht Auf(LoescheKarte l) => new(l.Station);
}

public sealed class Daumenmeldung : IFunktion
{
    public KarteBereit Auf(KarteSkaliert k) => new(k.Station, k.Breite);
}

public sealed record Kartenfilter(
    [Kopf] Kantenglaettung Glaetten,
    [Von(nameof(Glaetten))] Kartenschaerfung Schaerfen,
    [Von(nameof(Schaerfen))] Kantenglaettung Nachglaetten) : IGraph;

public sealed record Kartenausgabe(
    [Kopf] Kartenzeichnung Zeichnung,
    [Von(nameof(Zeichnung))] Kartenfilter Filter,
    [Von(nameof(Filter))] Kartenskalierung Vorschau,
    [Von(nameof(Filter))] Kartenskalierung Daumen,
    [Kopf, Von(nameof(Vorschau))] Kartenablage Ablage,
    [Von(nameof(Daumen))] Daumenmeldung Meldung) : IGraph;

// Draußen: der Warndienst meldet Temperaturen, bestellt und löscht Karten und hört die Warnung, die Stilllegung und die Karten (Hinweise:
// Kenntnis); die Wetter-App verkörpert ihn.
public sealed record Warndienst : IAkteur, IDarf<MeldeTemperatur>, IDarf<ZeichneKarte>, IDarf<LoescheKarte>, IHoert<Unwetter>,
    IHoert<StationStillgelegt>, IHoert<KarteBereit>, IHoert<KarteGeloescht>;

public sealed record WetterApp : IClient, IVerkoerpert<Warndienst>;
