using Abstractions;
using Warenkatalog.Import;
using Sondenkatalog.Katalog;

// SONDE — die Übersetzungen des Warenkatalogs (docs/04 §4.4): der Import an den Vorgang, seine ANTWORT in den Kopf des Zeilenimports,
// Fortschritt und Bilanz an den Vorgang, jeder Artikel an seinen Stream — Guid ⇄ Id ausdrücklich, der Rest nach Namen.

namespace Warenkatalog.Uebersetzungen;

#pragma warning disable CS9113

public sealed class Importieren_StarteImport_1(Importieren source) : ITranslate<Importieren, StarteImport>
{
    public StarteImport Translate() => new(AggregateId: new(source.Import), Pfad: source.Pfad);
}

public sealed class ImportGestartet_KatalogBereit_1(ImportGestartet source) : ITranslate<ImportGestartet, KatalogBereit>
{
    public KatalogBereit Translate() => new(Import: source.ImportId.Wert, Pfad: source.Pfad);
}

public sealed class Fortschritt_MeldeImportFortschritt_1(Fortschritt source) : ITranslate<Fortschritt, MeldeImportFortschritt>
{
    public MeldeImportFortschritt Translate() => new(AggregateId: new(source.Import), Zeilen: source.Zeilen);
}

public sealed class Bilanz_SchliesseImportAb_1(Bilanz source) : ITranslate<Bilanz, SchliesseImportAb>
{
    public SchliesseImportAb Translate() => new(AggregateId: new(source.Import), Zeilen: source.Zeilen, Fehler: source.Fehler);
}

public sealed class Artikeldaten_LegeArtikelAn_1(Artikeldaten source) : ITranslate<Artikeldaten, LegeArtikelAn>
{
    public LegeArtikelAn Translate() => new(AggregateId: new(source.Artikel), Nummer: source.Nummer, Bezeichnung: source.Bezeichnung, Preis: source.Preis);
}
