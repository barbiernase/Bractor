using Abstractions;

// SONDE — fremdes Szenario 1 (docs/12 §12.6): BENUTZER MIT EINDEUTIGER E-MAIL.
//
// „Keine zwei Benutzer mit derselben Adresse“ ist eine Invariante über viele Benutzer — sie gehört in EIN Aggregat: die Adresse selbst
// (Eindeutigkeits-Aggregat, Id aus der normalisierten Adresse abgeleitet). Der Benutzer beantragt; auf das Fakt gibt die Adressprüfung (ein Client)
// die Prüfung der Adresse als Command ohne Ziel hinein. Die Domäne ist rein: die Registrierung ist ein Graph des fremden Katalogs
// (Katalog.*.cs.txt, Teil Konten) — sein einziger Knoten reserviert die Adresse und hört die ANTWORTEN darauf, die ANTWORT-ÜBERSETZUNGEN aus den zwei Events des
// Adress-Aggregats machen (Uebersetzungen.Konten.cs.txt): reserviert → bestätigen, vergeben → ablehnen. Die Absage ist ein FAKT (der Modellierer
// entscheidet über den Typ), weil der Ablauf an ihr weitergeht. Die Zugehörigkeit zum Benutzer ist ein fachliches Feld im Event (docs/10 A2).

namespace Kundenportal.Konten;

public record RegistriereBenutzer(Id<Benutzer> AggregateId, string Email, string Name) : ICommand<Benutzer>, ICreationCommand;
public record RegistrierungBeantragt(Id<Benutzer> BenutzerId, string Email, string Name) : IEvent;
public record BenutzerExistiert(Id<Benutzer> Id) : ITransientEvent;

public record BestaetigeBenutzer(Id<Benutzer> AggregateId) : ICommand<Benutzer>;
public record BenutzerBestaetigt(string Email) : IEvent;

public record LehneRegistrierungAb(Id<Benutzer> AggregateId, string Grund) : ICommand<Benutzer>;
public record RegistrierungAbgelehnt(string Grund) : IEvent;

public record ReserviereAdresse(Id<Adresse> AggregateId, string Email, Id<Benutzer> BenutzerId) : ICommand<Adresse>;
public record AdresseReserviert(string Email, Id<Benutzer> BenutzerId) : IEvent;
public record AdresseVergeben(Id<Benutzer> BenutzerId, Id<Benutzer> Inhaber) : IEvent;

/// <summary>Der Kopf der Adressprüfung: die Adresse des neuen Benutzers ist zu prüfen — die Adressprüfung reagiert damit auf den Antrag.</summary>
public record PruefeAdresse(Id<Benutzer> BenutzerId, string Email) : ICommand;

public enum Kontostatus { Beantragt = 1, Aktiv = 2, Abgelehnt = 3 }

public partial class Benutzer : IState
{
    public string Email { get; set; } = "";
    public Kontostatus Status { get; set; }

    public partial class Decider : IDecider<Benutzer>
    {
        public IEnumerable<OneOf<RegistrierungBeantragt, BenutzerExistiert>> Decide(RegistriereBenutzer c)
        {
            if (this.State.Version > 0) { yield return new BenutzerExistiert(c.AggregateId); yield break; }
            yield return new RegistrierungBeantragt(c.AggregateId, Adressschluessel.Normalisiere(c.Email), c.Name);
        }

        public IEnumerable<OneOf<BenutzerBestaetigt>> Decide(BestaetigeBenutzer c)
        {
            if (this.State.Status != Kontostatus.Beantragt) yield break;
            yield return new BenutzerBestaetigt(this.State.Email);
        }

        public IEnumerable<OneOf<RegistrierungAbgelehnt>> Decide(LehneRegistrierungAb c)
        {
            if (this.State.Status != Kontostatus.Beantragt) yield break;
            yield return new RegistrierungAbgelehnt(c.Grund);
        }
    }

    public partial class Applier : IApplier<Benutzer>
    {
        public void Apply(RegistrierungBeantragt e) { this.State.Email = e.Email; this.State.Status = Kontostatus.Beantragt; }
        public void Apply(BenutzerBestaetigt e) { this.State.Status = Kontostatus.Aktiv; }
        public void Apply(RegistrierungAbgelehnt e) { this.State.Status = Kontostatus.Abgelehnt; }
    }
}

// Das Eindeutigkeits-Aggregat: je Adresse ein Stream, der erste Benutzer gewinnt; derselbe Benutzer noch einmal ist ein Noop.
public partial class Adresse : IState
{
    public Id<Benutzer> Inhaber { get; set; }

    public partial class Decider : IDecider<Adresse>
    {
        public IEnumerable<OneOf<AdresseReserviert, AdresseVergeben>> Decide(ReserviereAdresse c)
        {
            if (this.State.Inhaber == c.BenutzerId) yield break;
            if (this.State.Inhaber != default) { yield return new AdresseVergeben(c.BenutzerId, this.State.Inhaber); yield break; }
            yield return new AdresseReserviert(c.Email, c.BenutzerId);
        }
    }

    public partial class Applier : IApplier<Adresse>
    {
        public void Apply(AdresseReserviert e) { this.State.Inhaber = e.BenutzerId; }
        public void Apply(AdresseVergeben e) { }
    }
}

// Hilfscode ohne Baustein-Rolle: die Adresse wird normalisiert, bevor sie ins Fakt geht.
public static class Adressschluessel
{
    public static string Normalisiere(string email) => email.Trim().ToLowerInvariant();
}

// Wer hineingibt: ein Mensch am Registrierungs-Formular; er hört, wie es ausgeht (Kenntnis zweier Events). Der Client nennt nur ihn.
public sealed record Besucher : IAkteur, IDarf<RegistriereBenutzer>, IHoert<BenutzerBestaetigt>, IHoert<RegistrierungAbgelehnt>;

// Ein Dienst, der reagiert: auf jeden Antrag gibt er die Prüfung seiner Adresse an den Kopf der Adressprüfung.
public sealed record Adresspruefer : IAkteur, IHoert<RegistrierungBeantragt>, IDarf<PruefeAdresse>;

public sealed record Registrierungsseite : IClient, IVerkoerpert<Besucher>;
