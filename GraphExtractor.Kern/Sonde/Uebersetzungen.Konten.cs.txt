using Abstractions;
using Sondenkatalog.Katalog;

// SONDE — die Übersetzungen des Kundenportals (docs/04 §4.4): die Prüfung hinein, die Reservierung an die Adresse, die zwei ANTWORTEN
// der Adresse zurück in den Katalog (Antwort-Übersetzungen, Event als Eingang), Bestätigung und Absage an den Benutzer — Guid ⇄ Id ausdrücklich.

namespace Kundenportal.Konten;

#pragma warning disable CS9113

public sealed class PruefeAdresse_Reserviere_1(PruefeAdresse source) : ITranslate<PruefeAdresse, Reserviere>
{
    public Reserviere Translate() => new(Benutzer: source.BenutzerId.Wert, Email: source.Email);
}

public sealed class Reservierung_ReserviereAdresse_1(Reservierung source) : ITranslate<Reservierung, ReserviereAdresse>
{
    public ReserviereAdresse Translate() => new(AggregateId: new(source.Adresse), Email: source.Email, BenutzerId: new(source.Benutzer));
}

public sealed class AdresseReserviert_Reserviert_1(AdresseReserviert source) : ITranslate<AdresseReserviert, Reserviert>
{
    public Reserviert Translate() => new(Benutzer: source.BenutzerId.Wert);
}

public sealed class AdresseVergeben_Vergeben_1(AdresseVergeben source) : ITranslate<AdresseVergeben, Vergeben>
{
    public Vergeben Translate() => new(Benutzer: source.BenutzerId.Wert);
}

public sealed class Bestaetige_BestaetigeBenutzer_1(Bestaetige source) : ITranslate<Bestaetige, BestaetigeBenutzer>
{
    public BestaetigeBenutzer Translate() => new(AggregateId: new(source.Benutzer));
}

public sealed class Ablehnen_LehneRegistrierungAb_1(Ablehnen source) : ITranslate<Ablehnen, LehneRegistrierungAb>
{
    public LehneRegistrierungAb Translate() => new(AggregateId: new(source.Benutzer), Grund: source.Grund);
}
