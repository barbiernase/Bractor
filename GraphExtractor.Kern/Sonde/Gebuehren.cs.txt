using Abstractions;
using Leihwesen.Ausleihe;

// SONDE — ZWEI Aggregate im SELBEN Namespace, Decider in eigener Datei (Mahnvorgang.Decider.cs.txt). Die Domäne ist rein: der Ablauf zwischen
//         ihnen sind GRAPHEN des fremden Katalogs (Katalog.*.cs.txt, Teil Gebuehren), die Übersetzungen (Uebersetzungen.Leihwesen.cs.txt) binden sie:
//         · FOLGE mit leerem Strom: die Rückgabeprüfung ist ein Graph mit EINEM Knoten; sein Kopf nimmt die Prüfung eines Clients (PruefeRueckgabe,
//           übersetzt) und gibt auf einem Teil der Fälle nichts aus (ein leerer Strom); die neue Id kommt aus der Herkunft (R13);
//         · WAHL: die Schadensaufnahme gibt genau einen von drei Fällen aus — einen Command an den nächsten Knoten, die Begutachtung vor Ort (übersetzt
//           an den Schadensfall; auf das Fakt reagiert das Archiv) oder einen Hinweis (flüchtig, an die Akteure, die ihn hören);
//         · KETTE, DIE MITTRÄGT: die Schätzung (TECHNIK im Konstruktor) gibt ihren Betrag an das Gutachten (KONFIG-Record), das Gutachten trägt
//           Betrag und Zuschlag zur Mahnung — jedes Auf nimmt genau eine Nachricht, keiner muss zusammenführen.

namespace Leihwesen.Gebuehren;

public record MeldeSchaden(Id<Schadensfall> AggregateId, Guid LeserId) : ICommand<Schadensfall>, ICreationCommand;
public record SchadenGemeldet(Guid LeserId) : IEvent;
public record SchadenExistiert(Id<Schadensfall> Id) : ITransientEvent;

public record EroeffneMahnung(Id<Mahnvorgang> AggregateId, decimal Betrag) : ICommand<Mahnvorgang>, ICreationCommand;
public record MahnungEroeffnet(decimal Betrag) : IEvent;
public record MahnungExistiert(Id<Mahnvorgang> Id) : ITransientEvent;

/// <summary>Der Kopf der Schadensbewertung: der Schaden ist zu schätzen — die Gutachterin gibt ihn hinein, sie reagiert auf das gemeldete Fakt.</summary>
public record SchaetzeSchaden(Id<Schadensfall> Fall, Guid LeserId) : ICommand;

/// <summary>Die Begutachtung vor Ort (einer der drei Fälle der Schadensaufnahme, übersetzt an den Schadensfall).</summary>
public record BegutachteVorOrt(Id<Schadensfall> AggregateId, Guid LeserId) : ICommand<Schadensfall>;
public record BegutachtungAngefordert(Guid LeserId) : IEvent;

public partial class Schadensfall : IState
{
    public bool Gemeldet { get; set; }
    public bool Begutachtet { get; set; }

    public partial class Decider : IDecider<Schadensfall>
    {
        public IEnumerable<OneOf<SchadenGemeldet, SchadenExistiert>> Decide(MeldeSchaden c)
        {
            if (this.State.Gemeldet) { yield return new SchadenExistiert(c.AggregateId); yield break; }
            yield return new SchadenGemeldet(c.LeserId);
        }

        public IEnumerable<OneOf<BegutachtungAngefordert>> Decide(BegutachteVorOrt c)
        {
            if (this.State.Begutachtet) yield break;
            yield return new BegutachtungAngefordert(c.LeserId);
        }
    }

    public partial class Applier : IApplier<Schadensfall>
    {
        public void Apply(SchadenGemeldet e) { this.State.Gemeldet = true; }
        public void Apply(BegutachtungAngefordert e) { this.State.Begutachtet = true; }
    }
}

public partial class Mahnvorgang : IState
{
    public decimal Betrag { get; set; }
}
