using Abstractions;
using Leihwesen.Ausleihe;

// SONDE — ZWEI Aggregate im SELBEN Namespace + Saga mit expliziter Interface-Implementierung und Block-Lambda,
//         dazu eine KATALOG-FUNKTION (IFunktion) mit eigenem Auftrag, die die Saga per Rufe<F> mit Zeitlimit ruft — ihre C#-Implementierung
//         nimmt einen DOMÄNEN-DIENST (IDienst) im Konstruktor, dessen Implementierung einen Konfig-Record nimmt (konzept-werkbank §12.3 E11).

namespace Leihwesen.Gebuehren;

public record MeldeSchaden(Guid AggregateId, Guid LeserId) : ICreationCommand;
public record SchadenGemeldet(Guid LeserId) : IEvent;
public record SchadenExistiert(Guid Id) : ITransientEvent;

public record EroeffneMahnung(Guid AggregateId, decimal Betrag) : ICreationCommand;
public record MahnungEroeffnet(decimal Betrag) : IEvent;
public record MahnungExistiert(Guid Id) : ITransientEvent;

// Katalog-Funktion: der Auftrag bindet sich per IAuftrag<F> an genau sie; die Ergebnisse sind persistente Events.
public sealed record SchaetzeSchaden(Guid LeserId, decimal Neuwert) : IAuftrag<ISchadensSchaetzung>;
public record SchadenGeschaetzt(decimal Betrag) : IEvent;
public record NichtSchaetzbar(string Grund) : IEvent;

public interface ISchadensSchaetzung : IFunktion
{
    Task<OneOf<SchadenGeschaetzt, NichtSchaetzbar>> RufeAsync(SchaetzeSchaden auftrag, IAusfuehrung x);
}

// Domänen-Dienst: die Technik hinter der Funktion — ein Interface mit IDienst, eine Implementierung mit Konfig-Record (Werte aus Konfig:<Record>).
public interface IWertTabelle : IDienst
{
    /// <summary>Der Restwert nach linearer Abschreibung.</summary>
    decimal Restwert(decimal neuwert, int alterJahre);
    Task<bool> IstSchaetzbarAsync(Guid leserId);
}

public record WertTabellenKonfig(decimal Mindestwert, int Abschreibungsjahre = 5);

public sealed class LineareWertTabelle(WertTabellenKonfig konfig) : IWertTabelle
{
    public decimal Restwert(decimal neuwert, int alterJahre) => Math.Max(konfig.Mindestwert, neuwert * (1 - alterJahre / (decimal)konfig.Abschreibungsjahre));

    public Task<bool> IstSchaetzbarAsync(Guid leserId)
    {
        return Task.FromResult(leserId != Guid.Empty);
    }
}

public sealed class Schadensschaetzer(IWertTabelle tabelle) : ISchadensSchaetzung
{
    public async Task<OneOf<SchadenGeschaetzt, NichtSchaetzbar>> RufeAsync(SchaetzeSchaden auftrag, IAusfuehrung x)
    {
        if (!await tabelle.IstSchaetzbarAsync(auftrag.LeserId)) return new NichtSchaetzbar("unbekannter Leser");
        return new SchadenGeschaetzt(tabelle.Restwert(auftrag.Neuwert, 2));
    }
}

public partial class Schadensfall : IState
{
    public bool Gemeldet { get; set; }

    public partial class Decider : IDecider<Schadensfall>
    {
        public IEnumerable<OneOf<SchadenGemeldet, SchadenExistiert>> Decide(MeldeSchaden c)
        {
            if (this.State.Gemeldet) { yield return new SchadenExistiert(c.AggregateId); yield break; }
            yield return new SchadenGemeldet(c.LeserId);
        }
    }

    public partial class Applier : IApplier<Schadensfall>
    {
        public void Apply(SchadenGemeldet e) { this.State.Gemeldet = true; }
    }
}

public partial class Mahnvorgang : IState
{
    public decimal Betrag { get; set; }
}

public sealed class SchadenBeiRueckgabe : IProzessDefinition
{
    ProzessRegeln IProzessDefinition.Regeln => Prozess<Zurueckgegeben>.Definiere(p =>
    {
        p.Auf<Zurueckgegeben>()
         .Sende<MeldeSchaden>(e => new MeldeSchaden(Guid.NewGuid(), e.LeserId));

        p.Auf<SchadenGemeldet>()
         .Rufe<ISchadensSchaetzung>(s => new SchaetzeSchaden(s.LeserId, 25m))
         .Zeitlimit(TimeSpan.FromMinutes(5));

        p.Auf<SchadenGeschaetzt>()
         .Sende<EroeffneMahnung>(g => new EroeffneMahnung(Guid.NewGuid(), g.Betrag));
    });
}
