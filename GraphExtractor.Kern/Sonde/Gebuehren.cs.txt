using Abstractions;
using Leihwesen.Ausleihe;

// SONDE — ZWEI Aggregate im SELBEN Namespace, Decider in eigener Datei (Mahnvorgang.Decider.cs.txt), dazu der Ablauf zwischen ihnen aus
//         GRAPHEN (Entscheidungen 4–6; früher eine Saga mit expliziter Interface-Implementierung):
//         · FOLGE mit leerem Strom: die Rückgabeprüfung ist ein Graph mit EINEM Knoten; sein Kopf nimmt einen Command ohne Ziel aus dem Decide
//           (PruefeRueckgabe) und gibt auf einem Teil der Fälle nichts aus (ein leerer Strom); die neue Id kommt aus der Herkunft (R13);
//         · WAHL: die Schadensaufnahme gibt genau einen von drei Fällen aus — einen Command an den nächsten Knoten, Arbeit für einen Akteur draußen
//           (ein Command ohne Ziel, den ein Akteur hört) oder einen Hinweis (flüchtig, an die Akteure, die ihn hören);
//         · FORK: der erfasste Schaden geht an zwei Knoten (Schätzung mit TECHNIK im Konstruktor, Gutachten mit KONFIG-Record);
//         · JOIN mit WARTEN MIT FRIST: die Mahnung nimmt die Schätzung (genau eine) und das Gutachten als T? — höchstens eins: sie feuert, wenn es kam
//           oder das Zeitlimit ihres Knotens (Konfig der Instanz) abläuft; dann ist es null.

namespace Leihwesen.Gebuehren;

public record MeldeSchaden(Id<Schadensfall> AggregateId, Guid LeserId) : ICommand<Schadensfall>, ICreationCommand;
public record SchadenGemeldet(Guid LeserId) : IEvent;
public record SchadenExistiert(Id<Schadensfall> Id) : ITransientEvent;

public record EroeffneMahnung(Id<Mahnvorgang> AggregateId, decimal Betrag) : ICommand<Mahnvorgang>, ICreationCommand;
public record MahnungEroeffnet(decimal Betrag) : IEvent;
public record MahnungExistiert(Id<Mahnvorgang> Id) : ITransientEvent;

// Gleichbleibend: nur ein beschädigt zurückgegebenes Medium ist ein Schadensfall — je Rückgabe genau einer (Id aus der Herkunft).
public sealed class SchadenBeiRueckgabe : IFunktion
{
    public IEnumerable<MeldeSchaden> Auf(PruefeRueckgabe p, Herkunft h)
    {
        if (p.Zustand == Zustand.Beschaedigt) yield return new MeldeSchaden(h.NeueId<Schadensfall>(), p.LeserId);
    }
}

// Ein Command ohne Ziel aus dem Decide des Schadensfalls: der Schaden ist zu schätzen (der Kopf der Schadensbewertung).
public record SchaetzeSchaden(Id<Schadensfall> Fall, Guid LeserId) : ICommand;

// Was nur Knoten einander geben (Commands ohne Ziel, nie im Log) — und was draußen ankommt: Arbeit für einen Akteur, ein Hinweis.
public record SchadenErfasst(Guid LeserId, decimal Neuwert, int AlterJahre) : ICommand;
public record SchadenGeschaetzt(Guid LeserId, decimal Betrag) : ICommand;
public record GutachtenErhalten(Guid LeserId, decimal Zuschlag) : ICommand;
public record BegutachteVorOrt(Id<Schadensfall> Fall, Guid LeserId) : ICommand;
public record NichtSchaetzbar(string Grund) : ITransientEvent;

// WAHL: genau einer der drei Fälle je Lauf.
public sealed class Schadensaufnahme : IFunktion
{
    public OneOf<SchadenErfasst, BegutachteVorOrt, NichtSchaetzbar> Auf(SchaetzeSchaden s)
    {
        if (s.LeserId == Guid.Empty) return new NichtSchaetzbar("unbekannter Leser");
        if (s.Fall.Wert.ToByteArray()[0] % 2 == 0) return new BegutachteVorOrt(s.Fall, s.LeserId);
        return new SchadenErfasst(s.LeserId, 25m, 2);
    }
}

// Technik hinter der Schätzung — eine Schnittstelle ohne Marker (der Schätzer nimmt sie im Konstruktor), eine Umsetzung mit Konfig-Record
// (Werte am Knoten der Instanz).
public interface IWertTabelle
{
    /// <summary>Der Restwert nach linearer Abschreibung.</summary>
    decimal Restwert(decimal neuwert, int alterJahre);
    Task<bool> IstSchaetzbarAsync(Guid leserId);
}

public record WertTabellenKonfig(decimal Mindestwert, int Abschreibungsjahre = 5);

public sealed class LineareWertTabelle(WertTabellenKonfig konfig) : IWertTabelle
{
    public decimal Restwert(decimal neuwert, int alterJahre) => Math.Max(konfig.Mindestwert, neuwert * (1 - alterJahre / (decimal)konfig.Abschreibungsjahre));

    public Task<bool> IstSchaetzbarAsync(Guid leserId)
    {
        return Task.FromResult(leserId != Guid.Empty);
    }
}

// FORK, Zweig 1: die Schätzung mit Technik.
public sealed class Schadensschaetzer(IWertTabelle tabelle) : IFunktion
{
    public async Task<SchadenGeschaetzt> Auf(SchadenErfasst e)
    {
        var faktor = await tabelle.IstSchaetzbarAsync(e.LeserId) ? 1m : 0m;
        return new SchadenGeschaetzt(e.LeserId, faktor * tabelle.Restwert(e.Neuwert, e.AlterJahre));
    }
}

// FORK, Zweig 2: ein Gutachten, das dauern kann — sein Zuschlag aus dem Konfig-Record des Knotens.
public record GutachtenKonfig(decimal Zuschlag, TimeSpan Bedenkzeit);

public sealed class Gutachtenanfrage(GutachtenKonfig konfig) : IFunktion
{
    public async Task<GutachtenErhalten> Auf(SchadenErfasst e)
    {
        await Task.Delay(konfig.Bedenkzeit);
        return new GutachtenErhalten(e.LeserId, konfig.Zuschlag);
    }
}

// JOIN mit WARTEN MIT FRIST: die Schätzung kommt genau einmal, das Gutachten höchstens einmal (null nach dem Zeitlimit des Knotens).
public sealed class MahnungNachSchaetzung : IFunktion
{
    public EroeffneMahnung Auf(SchadenGeschaetzt g, GutachtenErhalten? gutachten, Herkunft h) =>
        new(h.NeueId<Mahnvorgang>(), g.Betrag + (gutachten?.Zuschlag ?? 0m));
}

// Die Graphen: Knoten als Parameter, Kanten aus den Typen (SchadenErfasst: Aufnahme → Schätzung und Gutachten; beide → Mahnung).
public sealed record Rueckgabepruefung(SchadenBeiRueckgabe Pruefung) : IGraph;
public sealed record Schadensbewertung(Schadensaufnahme Aufnahme, Schadensschaetzer Schaetzung, Gutachtenanfrage Gutachten, MahnungNachSchaetzung Mahnung) : IGraph;

public partial class Schadensfall : IState
{
    public bool Gemeldet { get; set; }

    public partial class Decider : IDecider<Schadensfall>
    {
        public IEnumerable<OneOf<SchadenGemeldet, SchaetzeSchaden, SchadenExistiert>> Decide(MeldeSchaden c)
        {
            if (this.State.Gemeldet) { yield return new SchadenExistiert(c.AggregateId); yield break; }
            yield return new SchadenGemeldet(c.LeserId);
            yield return new SchaetzeSchaden(c.AggregateId, c.LeserId);
        }
    }

    public partial class Applier : IApplier<Schadensfall>
    {
        public void Apply(SchadenGemeldet e) { this.State.Gemeldet = true; }
    }
}

public partial class Mahnvorgang : IState
{
    public decimal Betrag { get; set; }
}
